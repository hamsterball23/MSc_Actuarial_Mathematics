\begin{enumerate}

\item[\textbf{Q1.}] \ans{A, B, C.}
\begin{itemize}[nosep]
\item[A] Hitting time of a continuous adapted process at a fixed level: a stopping time.
\item[B] $B_t-(t-1)$ is continuous and adapted, so $T_2$ is the hitting time of level $0$ by this process (Problem 3.4(a)): a stopping time.
\item[C] Let $\tau=\inf\{t\ge0\mid B_t\ge3\}$, a stopping time. For a stopping time $\tau$ and constant $c>0$, $\tau+c$ is again a stopping time: for $t<c$, $\{\tau+c\le t\}=\varnothing\in\mathcal F_t$ (since $\tau\ge0$); for $t\ge c$, $\{\tau+c\le t\}=\{\tau\le t-c\}\in\mathcal F_{t-c}\subseteq\mathcal F_t$. (Contrast this with \emph{backward} shifts, as in Quiz B Q1(C) / the practice quiz, which are \emph{not} stopping times.)
\item[D] $T_4$ is a last-exit time (supremum of a time set within $[0,5]$): deciding $\{T_4\le t\}$ for $t<5$ requires knowing that $B$ has \emph{no} further visit to level $2$ in $(t,5]$ -- information not available at time $t$. Not a stopping time.
\end{itemize}

\item[\textbf{Q2.}] \ans{C.} By Problem 5.2, if $A\in\mathcal F_s$ and $T=a$ on $A$, $T=b$ on $A^c$, then $T$ is a stopping time precisely when $a\wedge b\ge s$. Here $A=\{B_4>0\}\in\mathcal F_4$, so we need $a\wedge b\ge4$. Only option C, $(a,b)=(4,6)$, satisfies this.

\item[\textbf{Q3.}] \ans{A.} This is the process of Problem 3.1(c) with $\theta=3$ (so $\theta^2/2=4.5$): $S$ is a submartingale if $\mu>0$ and a supermartingale if $\mu<0$ (and a genuine martingale if $\mu=0$). Hence $S$ is a supermartingale iff $\mu<0$.

\item[\textbf{Q4.}] \ans{A.} By Result 20 (Toolbox) / Problem 3.1(e): $x\mapsto(x-4)^+$ is convex (and non-decreasing where it matters), so $(B_t-4)^+$ is a submartingale since $B$ is a martingale. The other options fail: $x\mapsto x^3$ and $x\mapsto\sin x$ are not convex (indeed, by It\^o, $d(B_t^3)=3B_t^2\,dB_t+3B_t\,dt$ and $d(\sin B_t)=\cos(B_t)\,dB_t-\tfrac12\sin(B_t)\,dt$ both have sign-indefinite drift, so neither is a submartingale); $x\mapsto-x^2$ is concave, giving a supermartingale, not a submartingale; $x\mapsto e^{-x^2}$ is not convex.

\item[\textbf{Q5.}] \ans{A.} The components $B_t^{(1)},B_t^{(2)},B_t^{(3)}$ are independent, each with variance $t$, so $\Var(\lambda\cdot\boldsymbol B_t)=\sum_i\lambda_i^2\Var(B_t^{(i)})=t(\lambda\cdot\lambda)$. Here $\lambda\cdot\lambda=1^2+2^2+2^2=9$, so $\Var(X_t)=9t$.

\item[\textbf{Q6.}] \ans{A.} By Problem 3.3(b)(i), $\Cov(W_s^{(1)},B_t^{(2)})=\rho(s\wedge t)$. For $s\le t$, $s\wedge t=s$, so $\Cov(W_s^{(1)},B_t^{(2)})=0.6s$.

\item[\textbf{Q7.}] \ans{A.} By Result 17 (Toolbox) / Problem 4.3, for integrated Brownian motion $I_t=\int_0^tB_s\,ds$, $\Var(I_t)=t^3/3$.

\item[\textbf{Q8.}] \ans{A.} Using $\Cov(I_s,I_t)=\tfrac{s^2t}2-\tfrac{s^3}6$ for $s\le t$ (Result 17): $\Cov(I_1,I_2)=\tfrac{1\cdot2}2-\tfrac16=1-\tfrac16=\tfrac56$. Then $\Cov(I_1,I_2-I_1)=\Cov(I_1,I_2)-\Var(I_1)=\tfrac56-\tfrac13=\tfrac12$. (In particular this is nonzero, so -- unlike Brownian motion itself -- integrated Brownian motion does \emph{not} have independent increments.)

\item[\textbf{Q9.}] \ans{A.} $W$ is a standard Brownian motion (Result 18, time inversion), so $\Var(W_t)=t$ directly. (Check: $\Var(W_t)=t^2\Var(B_{1/t})=t^2\cdot\tfrac1t=t$.)

\item[\textbf{Q10.}] \ans{A.} Doob's $L^2$ maximal inequality (Result 19) gives $\E\big[\sup_{t\in[0,T]}M_t^2\big]\le4\,\E[M_T^2]$. With $T=4$, $\E[M_4^2]=5\cdot4=20$, so the bound is $4\cdot20=80$.

\item[\textbf{Q11.}] \ans{A.} $B_t\sim N(0,t)$, so by the even-moment formula $\E[X^{2n}]=(2n-1)!!\,\sigma^{2n}$ for $X\sim N(0,\sigma^2)$ (Problem 1.1(c)) with $\sigma^2=t$, $n=3$: $\E[B_t^6]=5!!\cdot t^3=15t^3$.

\item[\textbf{Q12.}] \ans{A, B, D.}
\begin{itemize}[nosep]
\item[A] $B_s$ is $\mathcal F_s$-measurable and $(B_t-B_s)^2$ is a function of the increment $B_t-B_s$, which is independent of $\mathcal F_s$. Hence $\E[B_s(B_t-B_s)^2]=\E[B_s]\cdot\E[(B_t-B_s)^2]=0\cdot(t-s)=0$ (Problem 3.4(c)).
\item[B] Writing $B_t=B_s+(B_t-B_s)$: $\E[B_s^2B_t]=\E[B_s^3]+\E[B_s^2(B_t-B_s)]=0+\E[B_s^2]\cdot0=0$ (Problem 3.4(d)).
\item[C] Writing $B_u=B_t+(B_u-B_t)$: $\E[(B_t-B_s)B_u]=\E[(B_t-B_s)B_t]+\E[(B_t-B_s)(B_u-B_t)]=\big(\E[(B_t-B_s)B_s]+\E[(B_t-B_s)^2]\big)+0=(0+(t-s))+0=t-s\ne0$ in general (Problem 3.4(e)). \textbf{Not} zero.
\item[D] $B_s,B_t,B_u$ are jointly Gaussian and mean zero; the third (odd) moment of their product vanishes: $\E[B_sB_tB_u]=0$ (Problem 3.4(f)).
\end{itemize}

\item[\textbf{Q13.}] \ans{A.} $X$ is a standard Brownian motion by Brownian scaling (Result 18), so $\Var(X_t)=t$ for every $t$; in particular $\Var(X_5)=5$. (Check directly: $\Var(X_5)=\lambda^2\Var(B_{5/\lambda^2})=9\cdot\tfrac59=5$.)

\item[\textbf{Q14.}] \ans{A.} By the It\^o isometry, $\E[X_s^2]=\int_0^se^{6u}\,du=\tfrac16(e^{6s}-1)$. Applying the isometry again to the outer integral,
\[
\E\left[\left(\int_0^tX_s\,dB_s\right)^2\right]=\int_0^t\E[X_s^2]\,ds=\int_0^t\tfrac16(e^{6s}-1)\,ds=\tfrac1{36}(e^{6t}-1)-\tfrac t6.
\]

\item[\textbf{Q15.}] \ans{D.} $X_t=e^{-3t}B_{e^{6t}}$ is a deterministic rescaling of Brownian motion at a deterministic time, hence centered Gaussian: $\E[X_t]=e^{-3t}\E[B_{e^{6t}}]=0$. For $0\le s<t$ (so $e^{6s}<e^{6t}$),
\[
\Cov(X_s,X_t)=e^{-3s-3t}\Cov(B_{e^{6s}},B_{e^{6t}})=e^{-3s-3t}\min(e^{6s},e^{6t})=e^{-3s-3t}e^{6s}=e^{3s-3t}=e^{-3(t-s)}.
\]

\item[\textbf{Q16.}] \ans{B.} $e^{\lambda X_t}=e^{\lambda B_t+3\lambda t}=e^{\lambda B_t-\lambda^2t/2}\cdot e^{(3\lambda+\lambda^2/2)t}$. Since $e^{\lambda B_t-\lambda^2t/2}$ is always a martingale, the product is a martingale iff the remaining deterministic exponential factor is constant, i.e.\ $3\lambda+\lambda^2/2=0$, giving $\lambda=0$ or $\lambda=-6$. So $e^{-6X_t}$ is a martingale.

\end{enumerate}
