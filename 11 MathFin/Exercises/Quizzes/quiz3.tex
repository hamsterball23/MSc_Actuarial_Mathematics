{\Large Practice Quiz C --- Brownian Motion, Martingales \& Stochastic Integrals}\\[8pt]

\vspace{4pt}
\noindent\textbf{Instructions.}
\begin{itemize}[nosep,leftmargin=*]
  \item Unless stated otherwise, $B=(B_t,t\ge0)$ is a standard Brownian motion and $(\mathcal F_t,t\ge0)$ is its (completed, right-continuous) canonical filtration.
  \item Each question is worth the number of points shown. Questions marked \textbf{(Select ONE)} have exactly one correct option; questions marked \textbf{(Select ALL that apply)} may have more than one correct option, and you must select all of them for full marks.
  \item No partial credit is given for multi-select questions unless stated otherwise by your instructor.
  \item You may use any result proved on the problem sheets or in the lecture notes without re-proving it. Several questions on this quiz require combining more than one result, or applying It\^o's formula directly rather than matching a memorised template.
\end{itemize}

\begin{enumerate}

\item[\textbf{Q1.}] \textbf{(6 pts, Select ALL that apply)} Let $T_1=\inf\{t\ge0\mid B_t=2\}$ and $T_2=\inf\{t\ge0\mid B_t=-3\}$, both stopping times. Which of the following are also stopping times?
\begin{enumerate}
  \item $T_1\wedge T_2$
  \item $T_1\vee T_2$
  \item $T_2-T_1$
  \item $(T_1\wedge4)+(T_2\wedge4)$
\end{enumerate}

\item[\textbf{Q2.}] \textbf{(6 pts, Select ONE)} Let $f(x,t)=x^2+at$ for a constant $a\in\R$, and let $Y_t=f(B_t,t)$. Using It\^o's formula, determine the value of $a$ for which $Y$ is a martingale.
\begin{enumerate}
  \item $a=-1$
  \item $a=1$
  \item $a=0$
  \item $a=2$
  \item $a=-2$
\end{enumerate}

\item[\textbf{Q3.}] \textbf{(6 pts, Select ONE)} Let $\boldsymbol B=(B^{(1)},B^{(2)})$ be a $2$-dimensional Brownian motion and let $\lambda,\mu\in\R^2$ satisfy $\lambda\cdot\lambda=1$, $\mu\cdot\mu=1$, and $\lambda\cdot\mu=\rho$ (not necessarily $0$). For $0\le s\le t$, determine $\Cov(\lambda\cdot\boldsymbol B_t,\ \mu\cdot\boldsymbol B_s)$.
\begin{enumerate}
  \item $\rho s$
  \item $\rho t$
  \item $s$
  \item $\rho\sqrt{st}$
  \item $\rho(s+t)/2$
\end{enumerate}

\item[\textbf{Q4.}] \textbf{(7 pts, Select ONE)} Let $\big(W_t,B_t\big)$ be a correlated two-dimensional Brownian motion with correlation parameter $\rho=0.5$, and recall that $M_t=W_tB_t-\rho t$ is a martingale (Problem 3.3(c)). Determine $\E\big[\langle M,M\rangle_t\big]$.
\begin{enumerate}
  \item $1.25\,t^2$
  \item $t^2$
  \item $1.5\,t^2$
  \item $0.25\,t^2$
  \item $2t^2$
\end{enumerate}

\item[\textbf{Q5.}] \textbf{(7 pts, Select ONE)} Let $I_t=\int_0^tB_s\,ds$ be integrated Brownian motion. Determine $P(I_3>3)$, expressed in terms of the standard normal c.d.f.\ $\Phi$.
\begin{enumerate}
  \item $1-\Phi(1)$
  \item $\Phi(1)$
  \item $1-\Phi(3)$
  \item $\Phi(3)$
  \item $1/2$
\end{enumerate}

\item[\textbf{Q6.}] \textbf{(7 pts, Select ONE)} Let $X_t=B_t^4-6\int_0^tB_s^2\,ds$. Using It\^o's formula, determine which of the following best describes $X$.
\begin{enumerate}
  \item A martingale
  \item A submartingale, but not a martingale
  \item A supermartingale, but not a martingale
  \item Neither a submartingale nor a supermartingale
  \item Cannot be determined without further information
\end{enumerate}

\item[\textbf{Q7.}] \textbf{(6 pts, Select ONE)} Let $M_t=\int_0^ts\,dB_s$. Using the It\^o isometry together with Doob's $L^2$ maximal inequality, determine the resulting upper bound for $\E\big[\sup_{t\in[0,3]}M_t^2\big]$.
\begin{enumerate}
  \item $36$
  \item $9$
  \item $18$
  \item $27$
  \item $4$
\end{enumerate}

\item[\textbf{Q8.}] \textbf{(6 pts, Select ONE)} Determine $\E\big[(B_t+2)^4\big]$.
\begin{enumerate}
  \item $3t^2+24t+16$
  \item $3t^2+16$
  \item $3t^2+24t$
  \item $t^2+24t+16$
  \item $3t^2+8t+16$
\end{enumerate}

\item[\textbf{Q9.}] \textbf{(6 pts, Select ONE)} Let $B^{(1)},B^{(2)}$ be independent Brownian motions and let $X_t=\int_0^t\cos(s)\,dB_s^{(1)}+\int_0^t\sin(s)\,dB_s^{(2)}$. Determine $\langle X,X\rangle_t$ and hence, by L\'evy's characterisation, identify $X$.
\begin{enumerate}
  \item $\langle X,X\rangle_t=t$; by L\'evy's characterisation, $X$ is a standard Brownian motion.
  \item $\langle X,X\rangle_t=t/2$; $X$ is not a standard Brownian motion.
  \item $\langle X,X\rangle_t=\tfrac12\sin(2t)+\tfrac t2$; $X$ is not a standard Brownian motion.
  \item $\langle X,X\rangle_t=\cos(2t)$; $X$ is not a standard Brownian motion.
  \item $\langle X,X\rangle_t=2t$; $X$ is not a standard Brownian motion.
\end{enumerate}

\item[\textbf{Q10.}] \textbf{(5 pts, Select ONE)} Define $W_t=tB_{1/t}$ for $t>0$, $W_0=0$ (time inversion; $W$ is again a standard Brownian motion). Determine $\Cov(B_1,W_2)$.
\begin{enumerate}
  \item $1$
  \item $2$
  \item $0.5$
  \item $0$
  \item $4$
\end{enumerate}

\item[\textbf{Q11.}] \textbf{(7 pts, Select ALL that apply)} For fixed $0\le s<t$, which of the following quantities equal $0$?
\begin{enumerate}
  \item $\E[B_s^3(B_t-B_s)]$
  \item $\E[B_t^3(B_t-B_s)]$
  \item $\E[(B_t-B_s)^3]$
  \item $\E[B_s^2(B_t-B_s)^2]$
\end{enumerate}

\item[\textbf{Q12.}] \textbf{(7 pts, Select ONE)} Let $M$ be a continuous local martingale with $M_0=0$ and deterministic, continuous, increasing quadratic variation $\langle M,M\rangle_t=g(t)=t^2$. For $0\le s<t$ and $\lambda\in\R$, determine $\E\big[\exp(\lambda(M_t-M_s))\mid\mathcal F_s\big]$.
\begin{enumerate}
  \item $\exp\big(\tfrac{\lambda^2}2(t^2-s^2)\big)$
  \item $\exp\big(\tfrac{\lambda^2}2(t-s)\big)$
  \item $\exp\big(\lambda^2(t^2-s^2)\big)$
  \item $\exp\big(\tfrac{\lambda}2(t^2-s^2)\big)$
  \item $\exp\big(\tfrac{\lambda^2}4(t^2-s^2)\big)$
\end{enumerate}

\item[\textbf{Q13.}] \textbf{(7 pts, Select ONE)} Let $\boldsymbol B$ be a $d$-dimensional Brownian motion, $\lambda\in\R^d$, and $Z_t=\exp\big(\lambda\cdot\boldsymbol B_t-\tfrac12|\lambda|^2t\big)$ (a martingale, Problem 3.2(c)). Let $C_t=Z_t^2$. Determine $\E[C_t]$.
\begin{enumerate}
  \item $e^{|\lambda|^2t}$
  \item $e^{2|\lambda|^2t}$
  \item $e^{-|\lambda|^2t}$
  \item $1$
  \item $e^{|\lambda|^2t/2}$
\end{enumerate}

\item[\textbf{Q14.}] \textbf{(6 pts, Select ONE)} Let $X_t=\int_0^te^{4s}\,dB_s$. Determine the value of $\displaystyle\E\left[\left(\int_0^tX_s\,dB_s\right)^2\right]$.
\begin{enumerate}
  \item $\dfrac1{64}(e^{8t}-1)-\dfrac t8$
  \item $\dfrac1{64}(e^{4t}-1)-\dfrac t8$
  \item $\dfrac1{64}(e^{8t}-1)-\dfrac t4$
  \item $\dfrac18(e^{8t}-1)-\dfrac t8$
  \item $\dfrac1{64}(e^{8t}-1)$
\end{enumerate}

\item[\textbf{Q15.}] \textbf{(6 pts, Select ONE)} Determine which type of process $X_t=e^{-4t}B_{e^{8t}}$ is.
\begin{enumerate}
  \item Centered Gaussian process with $\Cov(X_s,X_t)=e^{4(t-s)}$ for $0\le s<t$.
  \item Gaussian process with $\E[X_t]=e^{-4t}$ and $\Cov(X_s,X_t)=e^{4(t-s)}$ for $0\le s<t$.
  \item Gaussian process with $\E[X_t]=e^{-4t}$ and $\Cov(X_s,X_t)=e^{-4(t-s)}$ for $0\le s<t$.
  \item Centered Gaussian process with $\Cov(X_s,X_t)=e^{-4(t-s)}$ for $0\le s<t$.
  \item Gaussian process with $\E[X_t]=e^{-4t}$ and $\Cov(X_s,X_t)=e^{8(t-s)}$ for $0\le s<t$.
\end{enumerate}

\item[\textbf{Q16.}] \textbf{(5 pts, Select ONE)} Let $X_t=B_t+4t$. Determine which process is a martingale.
\begin{enumerate}
  \item $e^{-4X_t}$
  \item $e^{-8X_t}$
  \item $e^{4X_t}$
  \item $e^{8X_t}$
  \item $e^{-X_t}$
\end{enumerate}

\end{enumerate}
