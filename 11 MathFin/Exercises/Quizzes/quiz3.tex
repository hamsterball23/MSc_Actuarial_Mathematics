\vspace{4pt}
\noindent\textbf{Instructions.}
\begin{itemize}[nosep,leftmargin=*]
  \item Unless stated otherwise, $B=(B_t,t\ge0)$ is a standard Brownian motion and $(\mathcal F_t,t\ge0)$ is its (completed, right-continuous) canonical filtration.
  \item Each question is worth the number of points shown. Questions marked \textbf{(Select ONE)} have exactly one correct option; questions marked \textbf{(Select ALL that apply)} may have more than one correct option, and you must select all of them for full marks.
  \item No partial credit is given for multi-select questions unless stated otherwise by your instructor.
  \item You may use any result proved on the problem sheets or in the lecture notes without re-proving it.
\end{itemize}

\begin{enumerate}

\item[\textbf{Q1.}] \textbf{(6 pts, Select ALL that apply)} Which of the following random times are stopping times with respect to $(\mathcal F_t)$?
\begin{enumerate}
  \item $T_1=\inf\{t\ge 0 \mid B_t=5\}$
  \item $T_2=\inf\{t\ge 0 \mid B_t=t-1\}$
  \item $T_3=\big(\inf\{t\ge 0 \mid B_t\ge 3\}\big)+1$
  \item $T_4=\sup\{t\in[0,5]\mid B_t=2\}$
\end{enumerate}

\item[\textbf{Q2.}] \textbf{(6 pts, Select ONE)} Let $A\in\mathcal F_4$ and define the random time
\[
T=\begin{cases}a & \text{on } A\\ b & \text{on } A^c,\end{cases}
\]
where $A=\{B_4>0\}$. For which pair $(a,b)$ is $T$ a stopping time?
\begin{enumerate}
  \item $(a,b)=(4,2)$
  \item $(a,b)=(2,4)$
  \item $(a,b)=(4,6)$
  \item $(a,b)=(3,4)$
  \item $(a,b)=(4,3)$
\end{enumerate}

\item[\textbf{Q3.}] \textbf{(6 pts, Select ONE)} Let $S_t=\exp\big(3B_t+(\mu-4.5)t\big)$ for fixed $\mu\in\R$. For which condition on $\mu$ is $S$ a supermartingale?
\begin{enumerate}
  \item $\mu<0$
  \item $\mu>0$
  \item $\mu=4.5$
  \item $\mu\le4.5$
  \item All $\mu\in\R$
\end{enumerate}

\item[\textbf{Q4.}] \textbf{(6 pts, Select ONE)} Which of the following processes is a submartingale (using that $B$ is a martingale and a convex, increasing function of a martingale is a submartingale)?
\begin{enumerate}
  \item $(B_t-4)^+$
  \item $B_t^3$
  \item $\sin(B_t)$
  \item $-B_t^2$
  \item $e^{-B_t^2}$
\end{enumerate}

\item[\textbf{Q5.}] \textbf{(6 pts, Select ONE)} Let $\boldsymbol B=(B_t^{(1)},B_t^{(2)},B_t^{(3)})$ be a $3$-dimensional Brownian motion and let $\lambda=(1,2,2)\in\R^3$. Let $X_t=\lambda\cdot \boldsymbol B_t$. Determine $\Var(X_t)$.
\begin{enumerate}
  \item $9t$
  \item $3t$
  \item $t$
  \item $9t^2$
  \item $5t$
\end{enumerate}

\item[\textbf{Q6.}] \textbf{(6 pts, Select ONE)} Let $\big(W_t^{(1)},B_t^{(2)}\big)$ be a correlated two-dimensional Brownian motion with correlation parameter $\rho=0.6$ (as in Problem 3.3). For $0\le s\le t$, determine $\Cov\big(W_s^{(1)},B_t^{(2)}\big)$.
\begin{enumerate}
  \item $0.6s$
  \item $0.6t$
  \item $s$
  \item $0.36s$
  \item $0.6\sqrt{st}$
\end{enumerate}

\item[\textbf{Q7.}] \textbf{(6 pts, Select ONE)} Let $I_t=\int_0^tB_s\,ds$ be integrated Brownian motion. Determine $\Var(I_t)$.
\begin{enumerate}
  \item $t^3/3$
  \item $t^2/2$
  \item $t^3/6$
  \item $t^2$
  \item $t^3$
\end{enumerate}

\item[\textbf{Q8.}] \textbf{(7 pts, Select ONE)} With $I_t$ as in Q7, determine $\Cov(I_1,I_2-I_1)$.
\begin{enumerate}
  \item $1/2$
  \item $0$
  \item $1/3$
  \item $5/6$
  \item $1$
\end{enumerate}

\item[\textbf{Q9.}] \textbf{(6 pts, Select ONE)} Define $W_t=tB_{1/t}$ for $t>0$, $W_0=0$ (time inversion; $W$ is again a standard Brownian motion). Determine $\Var(W_t)$.
\begin{enumerate}
  \item $t$
  \item $1/t$
  \item $t^2$
  \item $1$
  \item $t^3$
\end{enumerate}

\item[\textbf{Q10.}] \textbf{(7 pts, Select ONE)} Let $M$ be a continuous martingale with $M_0=0$ and $\E[M_t^2]=5t$. Using Doob's $L^2$ maximal inequality, determine the resulting upper bound for $\E\big[\sup_{t\in[0,4]}M_t^2\big]$.
\begin{enumerate}
  \item $80$
  \item $20$
  \item $40$
  \item $5$
  \item $100$
\end{enumerate}

\item[\textbf{Q11.}] \textbf{(6 pts, Select ONE)} Determine $\E[B_t^6]$.
\begin{enumerate}
  \item $15t^3$
  \item $6t^3$
  \item $15t^2$
  \item $3t^3$
  \item $45t^3$
\end{enumerate}

\item[\textbf{Q12.}] \textbf{(6 pts, Select ALL that apply)} For fixed $0\le s<t<u$, which of the following quantities equal $0$?
\begin{enumerate}
  \item $\E[B_s(B_t-B_s)^2]$
  \item $\E[B_s^2B_t]$
  \item $\E[(B_t-B_s)B_u]$
  \item $\E[B_sB_tB_u]$
\end{enumerate}

\item[\textbf{Q13.}] \textbf{(6 pts, Select ONE)} Let $\lambda=3$ and $X_t=\lambda B_{t/\lambda^2}=3B_{t/9}$ (Brownian scaling; $X$ is again a standard Brownian motion). Determine $\Var(X_5)$.
\begin{enumerate}
  \item $5$
  \item $45$
  \item $5/9$
  \item $15$
  \item $25$
\end{enumerate}

\item[\textbf{Q14.}] \textbf{(7 pts, Select ONE)} Let $X_t=\int_0^te^{3s}\,dB_s$. Determine the value of $\displaystyle\E\left[\left(\int_0^tX_s\,dB_s\right)^2\right]$.
\begin{enumerate}
  \item $\dfrac1{36}(e^{6t}-1)-\dfrac t6$
  \item $\dfrac1{36}(e^{3t}-1)-\dfrac t6$
  \item $\dfrac1{36}(e^{6t}-1)-\dfrac t3$
  \item $\dfrac16(e^{6t}-1)-\dfrac t6$
  \item $\dfrac1{36}(e^{6t}-1)$
\end{enumerate}

\item[\textbf{Q15.}] \textbf{(7 pts, Select ONE)} Determine which type of process $X_t=e^{-3t}B_{e^{6t}}$ is.
\begin{enumerate}
  \item Centered Gaussian process with $\Cov(X_s,X_t)=e^{3(t-s)}$ for $0\le s<t$.
  \item Gaussian process with $\E[X_t]=e^{-3t}$ and $\Cov(X_s,X_t)=e^{3(t-s)}$ for $0\le s<t$.
  \item Gaussian process with $\E[X_t]=e^{-3t}$ and $\Cov(X_s,X_t)=e^{-3(t-s)}$ for $0\le s<t$.
  \item Centered Gaussian process with $\Cov(X_s,X_t)=e^{-3(t-s)}$ for $0\le s<t$.
  \item Gaussian process with $\E[X_t]=e^{-3t}$ and $\Cov(X_s,X_t)=e^{6(t-s)}$ for $0\le s<t$.
\end{enumerate}

\item[\textbf{Q16.}] \textbf{(6 pts, Select ONE)} Let $X_t=B_t+3t$. Determine which process is a martingale.
\begin{enumerate}
  \item $e^{-3X_t}$
  \item $e^{-6X_t}$
  \item $e^{3X_t}$
  \item $e^{6X_t}$
  \item $e^{-X_t}$
\end{enumerate}

\end{enumerate}
