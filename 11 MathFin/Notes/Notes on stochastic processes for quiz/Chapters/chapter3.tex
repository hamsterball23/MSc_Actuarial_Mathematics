\section{Chapter 3 --- Filtrations and Martingales}
% ==================================================================
Throughout this chapter we consider a probability space $(\Omega, \mc{F}, P)$.
\subsection{Filtrations and processes}
\begin{definition}[Filtration and filtered probability space]\label{chap3.1:definition-filtration}
\LG{Definition 3.1} \\
A \textit{filtration} on $(\Omega, \mc{F}, P)$ is a collection $(\mc{F}_t)_{0 \leq t \leq \infty}$ of sub-$\sigma$-algebras of $\mc{F}$ indexed by $[0,\infty]$, such that $\mc{F}_s \subseteq \mc{F}_t$ for every $0 \leq s \leq t \leq \infty$. We thus have, for every $0 \leq s < t$ $$\mc{F}_0 \subseteq \mc{F}_s \subseteq \mc{F}_t \subseteq \mc{F}_\infty \subseteq \mc{F}.$$
We also say that $(\Omega, \mc{F}, (\mc{F}_t), P)$ is a \textit{filtered probability space}.
\end{definition}

\begin{definition}[Canonical filtration]\label{chap3.1:definition-canonical_filtration}
\LG{p. 42} \\
Let $X = (X_t)_{t \geq 0}$ be any random process indexed by $\R_+$. The \textit{canonical filtration of $X$}  is defined by
$$\mc{F}_t^X:=\sigma(X_s, 0 \leq s \leq t)=\sigma\left(\bigcup_{s \leq t}\sigma(X_s)\right), \quad \mc{F}_\infty^X=\sigma(X_s, s \geq 0)=\sigma\left(\bigcup_{s \geq 0}\sigma(X_s)\right)$$
\end{definition}

\begin{definition}[Adaptedness]\label{chap3.1:definition-adaptedness}
\LG{Definition 3.3} \\
A random process $X=(X_t, t \geq 0)$ is called \textit{adapted} to a filtration $(\mc{F}_t)_{t \geq 0}$ if, for every $t \geq 0$, then $X_t$ is $\mc{F}_t$-measurable.
\end{definition}

Clearly $X$ is adapted to its canonical filtration by construction, which the following proposition makes clear.
\begin{proposition}\label{chap3.1:proposition-canonical_filtration_is_smallest_filtration_for_adaptness}
$(F^X_t)_{t \geq 0}$ is the smallest filtration to which $X$ is adapted. That is if $\mc{G}$ is any filtration that $X$ is adapted to, then $\mc{F}_t^X \subseteq \mc{G}$.
\end{proposition}
\begin{proof}
\textbf{Filtration property:} Let $s \leq t$. Then note
$$\bigcup_{u\leq s}\sigma(X_u) \subseteq\bigcup_{u \leq t} \sigma(X_u)$$
so every generator $X_u$ of $\mc{F}_s^X$ satisfies $u \leq s \leq t$, so it is also a generator of $\mc{F}_s^X$, which gives $\mc{F}_s^X \subseteq \mc{F}_t^X$. Also since $X_u$ is $\mc{F}$-measurable then $\mc{F}_t^X \subseteq \mc{F}$.\\
\textbf{Adaptedness:}. For each $t$, then $\sigma(X_t) \subseteq\mc{F}_t^X$ by construction, so therefore $X_t$ is $\mc{F}_t^X$-measurable.\\
\textbf{Minimality:} Let $(\mc{G}_t)$ be a filtration with $X$ adapted to it. Fix $t \geq 0$. For any $s \leq t$:
\begin{itemize}
    \item $X_s$ is $\mc{G}_s$-measurable, because $X$ is adapted to $(\mc{G}_t)$.
    \item $\mc{G}_s \subseteq \mc{G}_t$, because $(\mc{G}_t)$ is a filtration.
\end{itemize}
So $\sigma(X_s) \subseteq \mc{G}_t$ for every $s \leq t$. This means $\bigcup_{s \leq t}\sigma(X_s) \subseteq \mc{G}_t$. To see this last point then let $A \in \bigcup_{s \leq t}\sigma(X_s)$. Then there exists $0 \leq s' \leq t$ such that $A \in \sigma(X_{s'}) \subseteq\mc{G}$, which shows the containment.\\
Since $\mc{G}_t$ is a $\sigma$
-algebra, it also contains the smallest $\sigma$-algebra generated by that union that is
$$\mc{F}_t^X \subseteq \mc{G}_t$$
This shows the wanted minimality.
\end{proof}

\begin{definition}[Right-continuous filtration]\label{chap3.1:definition-right_cont_filtration} 
\LG{p. 42} \\
Let $(\mc{F})_{0 \leq t \leq \infty}$ be a filtration on $(\Omega, \mc{F}, P)$. For every $t \geq 0$, set
$$\mc{F}_{t+}=\bigcap_{s>t}\mc{F}_s$$
and $\mc{F}_{\infty+}=\mc{F}_\infty$. Note that $\mc{F}_t \subseteq \mc{F}_{t+}$ and that the collection $(\mc{F}_{t+})_{0 \leq t \leq \infty}$ is also a filtration. We say that the filtration $(\mc{F}_t)$ is \textit{right-continous} if
$$\mc{F}_{t+}=\mc{F}_t, \quad \forall t \geq 0.$$
By construction, the filtration $(\mc{F}_{t+})$ is right-continous.
\end{definition}

\begin{definition}[Completeness of filtrations]\label{chap3.1:definition-completeness_filtration}
\LG{p. 42} \\
Let $(\mc{F}_t)$ be a filtration and let $\mc{N}$ be the class of all $(\mc{F}_\infty,P)$-negligible sets, that is
$$A \in \mc{N} \quad \Leftrightarrow \quad \exists A' \in \mc{F}_\infty \text{ such that } A \subseteq A' \text{ and } P(A')=0$$
The filtration is said to be \textit{complete} if $\mc{N} \subseteq \mc{F}_0$ and thus $\mc{N} \subseteq \mc{F}_t$ for every $t \in [0,\infty]$.
\end{definition}
We will usually assume that the filtration we are working with is \textit{complete and right-continous}. This will ensure path regularity, and make stopping times and martingales well-behaved.
Without the completeness-assumptions we would not be able to infer that quantities like $\sup_{s \leq t}X_s$ are measurable, and one might end up with an event that is not contained in the $\sigma$-algebra.

Likewise right-continuity of filtrations ensures that hitting times of open sets is a stopping time, because such a hitting time requires looking an infinitesimal amount of time ahead. 

\textbf{One can always consider the completion of the filtration, and thereby safely avoid issues related to nasty null-sets in the following way:}
\begin{remark}[Completing a non-complete filtration] \label{chap3.1:remark-completing_a_non_complete_filtration}
\LG{p. 42} \\
For a given filtraation $(\mc{F}_t)_{t \geq 0}$ it possible to consider its \textit{completion} $\mc{F}'_t=\sigma\left(\mc{N} \cup \mc{F}_t\right)$, where $\mc{N}$ again is the collection of negligible elements of $\mc{F}_\infty$. Often we would apply this completion procedure to the canonical filtration of a random process, and call the resulting filtration the \textit{completed canonical filtration}.
\end{remark}

\begin{trap}\label{chap3.1:trap-right_continous_process_does_not_imply_right_continous_filtration}
The canonical filtration of a right-continuous process is \textbf{not right-continuous} in general. A counter example found in \cite{chafai2021completeness} is given by $X_t=tZ$ for all $t \geq 0$ where $Z$ is a non-constant random variable. Indeed, we have $\sigma(X_0)=\{\emptyset, \Omega\}$ while $\sigma(X_{0+\epsilon}: \epsilon > 0) = \sigma(Z) \neq \sigma(X_0)$. However it can be shown that the completion of the canonical filtration of a Feller-Markov process, including all Lévy processes and in particular Brownian motion, is always right-continuous.
\end{trap}


\begin{definition}[Measurable and progressive processes] \label{chap3.1:definition-measurable_and_progressive_processes}
\LG{Definitions 3.2 and 3.3}
A random process $(X_t)_{t \geq 0}$ with values in a measurable space
$(E,\mc{E})$ is said to be \textit{measurable} if the mapping

$$(\omega,s) \mapsto X_s(\omega)$$

defined on $\Omega \times \R_+$ equipped with the product
$\sigma$-algebra $\mc{F} \otimes \mc{B}(\R_+)$ is measurable.

A process is said to be \textit{progressive} or
\textit{progressively measurable} if, for every $t \geq 0$, the mapping

$$(\omega,s) \mapsto X_s(\omega)$$

defined on $\Omega \times [0,t]$ is measurable with respect to
$\mc{F}_t \otimes \mc{B}([0,t])$.
\end{definition}
\begin{proposition}[Local characterization of measurability]
\LG{p. 43, stated in first paragraph.} \\
A process $(X_t)_{t\geq0}$ is measurable if and only if, for every
$t\geq0$, the mapping
\[
(\omega,s)\mapsto X_s(\omega)
\]
restricted to $\Omega\times[0,t]$ is measurable with respect to
\[
\mc{F}\otimes\mc{B}([0,t]).
\]
\end{proposition}

\begin{proof}
The forward implication is immediate. Suppose that
$$(\omega,s)\mapsto X_s(\omega)$$
is measurable on $\Omega\times\mathbb R_+$ with respect to
$\mathcal F\otimes\mathcal B(\mathbb R_+)$. For any $t\geq0$, the set
$\Omega\times[0,t]$ is measurable, since $[0,t]\in\mathcal B(\mathbb R_+)$.
Hence, restricting the mapping to $\Omega\times[0,t]$ gives a measurable
mapping with respect to
$$\mathcal F\otimes\mathcal B([0,t]).$$
Conversely, suppose that for every $t\geq0$ the restricted mapping
$$(\omega,s)\mapsto X_s(\omega),
\qquad (\omega,s)\in\Omega\times[0,t],$$
is $\mc{F}\otimes\mc{B}([0,t])$-measurable.
Let $A\in\mc{E}$. For every $n\in\N$, we then have
$$\{(\omega,s)\in\Omega\times[0,n]:X_s(\omega)\in A\}
\in
\mc{F}\otimes\mc{B}([0,n]).$$
Since
$$\mc{F}\otimes\mc{B}([0,n])
\subseteq
\mc{F}\otimes\mc{B}(\R_+),$$
each of these sets belongs to $\mc{F}\otimes\mc{B}(\R_+)$.
Moreover,
$$\{(\omega,s)\in\Omega\times\R_+:X_s(\omega)\in A\}
=
\bigcup_{n=1}^{\infty}
\{(\omega,s)\in\Omega\times[0,n]:X_s(\omega)\in A\}.$$
Since $\mc{F}\otimes\mc{B}(\R_+)$ is a $\sigma$-algebra, the right-hand
side belongs to $\mc{F}\otimes\mc{B}(\R_+)$.

Thus $(\omega,s)\mapsto X_s(\omega)$ is measurable.
\end{proof}

\begin{remark}[Progressive $\subseteq$ adapted + measurable] \label{chap3.1:remark-progressive_implies_adapted_plus_measurable}
A progressive process is both adapted and measurable.
Indeed, fix $t\geq0$. By progressive measurability, the mapping
$$
F_t:\Omega\times[0,t]\to E,
\qquad
F_t(\omega,s)=X_s(\omega),
$$
is $\mathcal F_t\otimes\mathcal B([0,t])$-measurable. Consider the mapping
$$
i_t:\Omega\to\Omega\times[0,t],
\qquad
i_t(\omega)=(\omega,t).
$$
This mapping is $\mathcal F_t$-measurable, and
$$
X_t=F_t\circ i_t.
$$
Hence $X_t$ is $\mathcal F_t$-measurable. Since $t$ was arbitrary, $(X_t)_{t\geq0}$ is adapted. Moreover, since $\mathcal F_t\subseteq\mathcal F$,
$$
\mathcal F_t\otimes\mathcal B([0,t])
\subseteq
\mathcal F\otimes\mathcal B([0,t]).
$$
Thus the process satisfies the local characterization of measurability, and is therefore measurable.
\end{remark}


\begin{proposition} \label{chap3.1:proposition-right_continuity_implies_progressivity}
\LG{Proposition 3.4}
Let $(X_t)_{t \geq 0}$ be a random process with values in a metric space $(E, \mc{E})$ (equipped with its Borel $\sigma$-algebra). Suppose that $X$ is adapted and that the sample paths of $X$ are right-continous. That is for every $\omega \in \Omega, t \mapsto X_t(\omega)$ is right-continous. Then $X$ is progressive and this holds if one replaces right-continuity by left continuity.
\end{proposition}
\begin{proofidea}
The right-continuous version:
\begin{enumerate}
    \item For every $n \geq 1$ and $s \in [0,t]$ define $X_s^n=\begin{cases}X_{kt/n}, \quad &s \in [(k-1)t/n, kt/n), k \in \{1,...,n\} \\ X_t, \quad &s = t \end{cases}$
    \item Note that right continuity of sample paths ensure that $X_s(\omega)=\lim_{n \to \infty} X_s^n(\omega)$ for every $s \in [0,t], \omega \in \Omega$.
    \item Show that $X_s^n(\omega)$ is $\mc{F}_t \otimes \mc{B}([0,t])$-measurable.
    \item Use that a pointwise limit of measurable functions is also measurable with the same measurability condition to conclude that $X$ is progressive.
\end{enumerate}
\end{proofidea}
\begin{remark}\label{chap3.1:remark-right_continuity_implies_progressivity_implies_adapted_plus_measurable}
Note that Proposition \ref{chap3.1:proposition-right_continuity_implies_progressivity} and Remark \ref{chap3.1:remark-progressive_implies_adapted_plus_measurable} shows that (under weak assumption on the metric space that $X$ attains values in)
$$\text{Adapted + right-continous sample paths} \implies \text{Progressive} \implies \text{Measurable and adapted}$$
\end{remark}

\begin{quizbox}
The implication chain $\text{Adapted}+\text{right- (or left-) continuous paths} \Rightarrow \text{Progressive} \Rightarrow \text{Adapted}+\text{Measurable}$ is the single most-reused fact of this subsection: every process we actually work with (BM, càdlàg martingales, stochastic-integral integrands) is progressive \emph{because} it is adapted with one-sided-continuous paths, never by checking the definition directly. \\
Also know \emph{why} we assume the \textit{usual conditions} (complete and right-continuous filtration): completeness makes running suprema like $\sup_{s\le t}X_s$ genuinely $\mc F_t$-measurable (needed for Doob's inequalities and OST) even though the sup is over an uncountable set, while right-continuity of $(\mc F_t)$ is exactly what makes hitting times of \emph{open} sets stopping times without needing the $\mc F_{t+}$ workaround (\cref{chap3.2:proposition-hitting_times} below).
\end{quizbox}

\begin{proposition}[The progressive $\sigma$-algebra] \label{chap3.1:proposition-progressive_sigma_algebra}
\LG{p. 43} \\
The collection $\mc{P}$ of all sets $A \in \mc{F} \otimes \mc{B}(\R_+)$ such that the process $X_t(\omega)=1_A(\omega,t)$ is progressive forms a $\sigma$-algebra on $\Omega \times \R_+$, which is called \textit{the progressive $\sigma$-algebra}. \\
A subset $A \subseteq \Omega \times \R_+$ belongs to $\mc{P}$ if and only if, for every $t \geq 0$, then $A \cap (\Omega \times [0,t])$ belongs to $\mc{F}_t \otimes \mc{B}([0,t])$.
\end{proposition}

\begin{proposition} \label{chap3.1:proposition-progressivity_from_sigma_algebra_characterization}
\LG{p. 43, stated in the bottom} \\
A process $(X_t)_{t \geq 0}$ is progressive if and only if the mapping 
$$(\omega,t) \mapsto X_t(\omega)$$
is measurable on $\Omega \times \R_0$ equipped with the $\sigma$-algebra $\mc{P}$.
\end{proposition}
\begin{proof}
Suppose first that $(X_t)_{t\geq0}$ is progressive. Let
$B\in\mc E$. For every $t\geq0$, progressive measurability implies that
the mapping
$$
(\omega,s)\mapsto X_s(\omega)
$$
restricted to $\Omega\times[0,t]$ is
$\mc F_t\otimes\mc B([0,t])$-measurable. Hence
$$
\begin{aligned}
X^{-1}(B)\cap(\Omega\times[0,t])
&=
\{(\omega,s)\in\Omega\times[0,t]:X_s(\omega)\in B\}\\
&\in
\mc F_t\otimes\mc B([0,t]).
\end{aligned}
$$
Since this holds for every $t\geq0$, the characterization of the
progressive $\sigma$-algebra gives
$$
X^{-1}(B)\in\mc P.
$$
Thus $X$ is $\mc P$-measurable.

Conversely, suppose that $X$ is $\mc P$-measurable. Let $t\geq0$ and
$B\in\mc E$. Then
$$
X^{-1}(B)\in\mc P.
$$
By the characterization of $\mc P$,
$$
X^{-1}(B)\cap(\Omega\times[0,t])
\in
\mc F_t\otimes\mc B([0,t]).
$$
But
$$
X^{-1}(B)\cap(\Omega\times[0,t])
=
(X|_{\Omega\times[0,t]})^{-1}(B).
$$
Therefore the restriction
$$
X|_{\Omega\times[0,t]}
:
\Omega\times[0,t]\to E
$$
is $\mc F_t\otimes\mc B([0,t])$-measurable. Since $t$ was arbitrary,
$(X_t)_{t\geq0}$ is progressive.
\end{proof}

\subsubsection{Measurability of running supremum from completeness}
% Requires: amsmath, amssymb, amsthm with proposition/remark environments
This subsection is adapted from an almost identical theorem from \cite{chafai2021completeness}.
\begin{proposition}[Measurability of the running supremum]
Let $(\Omega,\mathcal F,\mathbb P)$ be a probability space and let
$(X_t)_{t\in\mathbb R_+}$ be a process with values in a topological space $E$
equipped with its Borel $\sigma$-field $\mathcal E$, such that $\mathbb P$-a.s.\
the path $t\mapsto X_t$ is continuous. Let $f\colon E\to\mathbb R$ be
continuous (in particular $\mathcal E$-measurable), and set
\[
  S_t := \sup_{s\in[0,t]} f(X_s) \in (-\infty,+\infty], \qquad t\in\mathbb R_+ .
\]
\begin{enumerate}
  \item If $(\Omega,\mathcal F,\mathbb P)$ is complete, then $S_t$ is
        $\mathcal F$-measurable for every $t\in\mathbb R_+$.
  \item If $X$ is adapted to a filtration $(\mathcal F_t)_{t\in\mathbb R_+}$
        which is complete, i.e.\ $(\Omega,\mathcal F,\mathbb P)$ is complete and
        $\mathcal F_0$ contains all $\mathbb P$-negligible sets, then
        $(S_t)_{t\in\mathbb R_+}$ is adapted to $(\mathcal F_t)_{t\in\mathbb R_+}$.
\end{enumerate}
\end{proposition}

\begin{proof}
Let $\Omega'\in\mathcal F$ with $\mathbb P(\Omega')=1$ be an event on which
$t\mapsto X_t(\omega)$ is continuous. Fix $t\in\mathbb R_+$ and put
$D_t := ([0,t]\cap\mathbb Q)\cup\{t\}$, a countable dense subset of $[0,t]$.

\emph{Step 1: reduction to a countable supremum.}
Fix $\omega\in\Omega'$ and let $g(s):=f(X_s(\omega))$ for $s\in[0,t]$. As the
composition of the continuous maps $s\mapsto X_s(\omega)$ and $f$, the function
$g$ is continuous on $[0,t]$. Clearly $\sup_{D_t} g\le\sup_{[0,t]} g$.
Conversely, for $s\in[0,t]$ pick $q_n\in D_t$ with $q_n\to s$; by continuity,
\[
  g(s)=\lim_{n\to\infty} g(q_n)\le\sup_{q\in D_t} g(q).
\]
Hence, for all $\omega\in\Omega'$,
\[
  S_t(\omega)=\widetilde S_t(\omega),
  \qquad\text{where } \widetilde S_t:=\sup_{q\in D_t} f(X_q).
\]

\emph{Step 2: proof of (1).}
Each $f(X_q)$ is $\mathcal F$-measurable, since $X_q$ is
$\mathcal F/\mathcal E$-measurable and $f$ is $\mathcal E$-measurable. As a
countable supremum, $\widetilde S_t$ is $\mathcal F$-measurable with values in
$(-\infty,+\infty]$. Let $A\in\mathcal B((-\infty,+\infty])$. By Step 1,
\[
  \{S_t\in A\}
  =\bigl(\Omega'\cap\{\widetilde S_t\in A\}\bigr)
   \,\cup\,\bigl((\Omega\setminus\Omega')\cap\{S_t\in A\}\bigr).
\]
The first set lies in $\mathcal F$, because $\tilde{S}_t$ is a countable supremum of $\mc{F}$-measurable maps, making the supremum also $\mc{F}$-measurable. The second is a subset of the
$\mathbb P$-null set $\Omega\setminus\Omega'$, hence belongs to $\mathcal F$ by
completeness. Thus $\{S_t\in A\}\in\mathcal F$.

\emph{Step 3: proof of (2).}
Fix $t$. For every $q\in D_t$ we have $q\le t$, so $X_q$ is
$\mathcal F_q$-measurable and $\mathcal F_q\subseteq\mathcal F_t$; hence
$f(X_q)$, and therefore $\widetilde S_t$, is $\mathcal F_t$-measurable.
Since $\Omega\setminus\Omega'$ is $\mathbb P$-null, it lies in
$\mathcal F_0\subseteq\mathcal F_t$, so $\Omega'\in\mathcal F_t$ and
$\Omega'\cap\{\widetilde S_t\in A\}\in\mathcal F_t$. The set
$(\Omega\setminus\Omega')\cap\{S_t\in A\}$ is $\mathbb P$-negligible, hence in
$\mathcal F_0\subseteq\mathcal F_t$. By the same decomposition as in Step 2,
$\{S_t\in A\}\in\mathcal F_t$. As $t$ was arbitrary, $(S_t)$ is adapted.
\end{proof}

\begin{remark}
\leavevmode
\begin{enumerate}
  \item On $\Omega'$, $S_t$ is in fact finite: $X([0,t])(\omega)$ is compact as
        the continuous image of a compact set, so $f$ is bounded on it.
  \item If \emph{every} path is continuous, one may take $\Omega'=\Omega$ and
        completeness is not needed.
  \item Continuity of $f$ cannot simply be dropped from the proof.
        With $U\sim\mathrm{Unif}[0,1]$, $X_s=s-U$, $f=\mathbf 1_{\{0\}}$ and
        $t=1$, we have $S_1\equiv 1$, whereas
        $\sup_{q\in D_1}f(X_q)=\mathbf 1_{\{U\in\mathbb Q\}}=0$ a.s.
  \item The conclusion nevertheless remains true for Borel $f$. For
        $c\in\mathbb R$,
        \[
          \{S_t>c\}\cap\Omega'
          =\pi_\Omega\bigl(\{(s,\omega)\in[0,t]\times\Omega' :
             f(X_s(\omega))>c\}\bigr),
        \]
        the projection of a $\mathcal B([0,t])\otimes\mathcal F_t$-measurable
        set (continuous adapted processes are progressive). By the measurable
        projection theorem this projection is $\mathcal F_t$-measurable when the
        filtration is complete, although in general it is only analytic
        (Lusin--Suslin).
\end{enumerate}
\end{remark}

\subsection{Stopping times and associated \texorpdfstring{$\sigma$}{sigma}-algebras}
This subsection lifts the stopping-time machinery of \cref{chap2.4:definition-stopping_time} from the Brownian canonical filtration to an arbitrary filtered probability space, and settles the two loose ends that right-continuity of $(\mc F_t)$ was advertised to fix in \cref{chap3.1:remark-completing_a_non_complete_filtration}: a clean characterisation of stopping times of $(\mc F_{t+})$, and hitting times of open sets.

\begin{definition}[Stopping time; $\mc F_T$ and $\mc F_{T+}$] \label{chap3.2:definition-stopping_time} \LG{Definition 3.5, p.\ 44--45} \\
A random variable $T:\Omega\to[0,\infty]$ is a \textit{stopping time} (of $(\mc F_t)$) if $\{T\le t\}\in\mc F_t$ for every $t\ge0$; as before, this gives $\{T<t\}\in\mc F_t$ for $t>0$ and $\{T=\infty\}\in\mc F_\infty$. The \textit{$\sigma$-field of the past before $T$} is
$$\mc F_T=\{A\in\mc F_\infty: \forall t\ge0,\; A\cap\{T\le t\}\in\mc F_t\},$$
a $\sigma$-field containing $T$ as a measurable function. Replacing $(\mc F_t)$ by the right-continuous filtration $(\mc F_{t+})$ throughout gives the corresponding notions of a stopping time \textit{of $(\mc F_{t+})$} and of the $\sigma$-field $\mc F_{T+}$; every $(\mc F_t)$-stopping time is an $(\mc F_{t+})$-stopping time, but not conversely in general --- this gap is exactly what the next proposition controls.
\end{definition}

\begin{proposition}[Stopping times of the right-continuous filtration] \label{chap3.2:proposition-stopping_time_of_right_cont_filtration} \LG{Proposition 3.6 (i)} \\
A random variable $T$ is a stopping time of $(\mc F_{t+})$ if and only if $\{T<t\}\in\mc F_t$ for every $t>0$, if and only if $T\wedge t$ is $\mc F_t$-measurable for every $t\ge0$.
\end{proposition}
\begin{proofidea}
If $T$ is a stopping time of $(\mc G_t):=(\mc F_{t+})$, write $\{T<t\}=\bigcup_{q\in\mathbb Q_+,q<t}\{T\le q\}$: each term lies in $\mc G_q\subseteq\mc F_t$ (since $q<t$), so the countable union does too. Conversely, if $\{T<t\}\in\mc F_t$ for every $t>0$, then for $s\ge0$, $\{T\le s\}=\bigcap_{q\in\mathbb Q_+,\,s<q<s+\epsilon}\{T<q\}\in\mc F_{s+\epsilon}$ for every $\epsilon>0$, hence $\{T\le s\}\in\mc F_{s+}=\mc G_s$. This is exactly the same dyadic/rational-sandwiching trick already used in \cref{chap2.4:definition-stopping_time} to recognise $T_a$ as a stopping time, run in both directions. The equivalence with $\mc F_t$-measurability of $T\wedge t$ is immediate by writing $\{T\wedge t\le s\}$ as $\{T\le s\}$ or $\Omega$ according as $s<t$ or $s\ge t$, and using the first equivalence on a sequence $s\uparrow t$.
\end{proofidea}

\begin{properties}[Algebra of stopping times] \label{chap3.2:properties-algebra_of_stopping_times} \LG{pp.\ 45--47} \\
Let $S,T$ (and a sequence $(T_n)$) be stopping times.
\begin{enumerate}[(a)]
    \item $\mc F_T\subseteq\mc F_{T+}$, with equality if $(\mc F_t)$ is right-continuous; a constant $T\equiv t$ gives back $\mc F_T=\mc F_t$.
    \item If $S\le T$, then $\mc F_S\subseteq\mc F_T$ (and likewise for the $+$ versions): for $A\in\mc F_S$, $A\cap\{T\le t\}=(A\cap\{S\le t\})\cap\{T\le t\}\in\mc F_t$.
    \item $S\wedge T$ and $S\vee T$ are again stopping times, since $\{S\wedge T\le t\}=\{S\le t\}\cup\{T\le t\}$ and $\{S\vee T\le t\}=\{S\le t\}\cap\{T\le t\}$ both lie in $\mc F_t$; moreover $\mc F_{S\wedge T}=\mc F_S\cap\mc F_T$, and $\{S\le T\}, \{S=T\}\in\mc F_{S\wedge T}=\mc F_S\cap\mc F_T$.
    \item If $T_n\uparrow T$ (non-decreasing), $T$ is a stopping time, since $\{T\le t\}=\bigcap_n\{T_n\le t\}$.
    \item If $T_n\downarrow T$ (non-increasing), $T$ is a stopping time of $(\mc F_{t+})$ --- using $\{T<t\}=\bigcup_n\{T_n<t\}\in\mc F_t$ and \cref{chap3.2:proposition-stopping_time_of_right_cont_filtration} --- and $\mc F_{T+}=\bigcap_n\mc F_{T_n+}$. If in addition $(T_n)$ is \emph{eventually stationary at $T$} (for a.e.\ $\omega$, $T_n(\omega)=T(\omega)$ for $n$ large), then $T$ is already a stopping time of $(\mc F_t)$ itself, and $\mc F_T=\bigcap_n\mc F_{T_n}$.
    \item (Le Gall's property (j), p.\ 47) A function $Y$ defined only on $\{T<\infty\}$ is $\mc F_T$-measurable if and only if, for every $t\ge0$, the restriction of $Y$ to $\{T\le t\}$ is $\mc F_t$-measurable. ($\Leftarrow$: $\{Y\in A\}\cap\{T\le t\}\in\mc F_t$ for every $t$ is exactly the defining property of $\mc F_T$. $\Rightarrow$: restrict the defining property to $\{T\le t\}$.) This is the criterion used just below to show $X_T$ is $\mc F_T$-measurable.
\end{enumerate}
In particular, for every stopping time $T$ the dyadic approximants
$$T_n=\sum_{k=0}^\infty\frac{k+1}{2^n}\ind_{\{k2^{-n}<T\le(k+1)2^{-n}\}}+\infty\cdot\ind_{\{T=\infty\}}, \qquad n\ge0,$$
form a sequence of stopping times decreasing to $T$; this is the discretisation device used repeatedly in \cref{chap2.4:theorem-strong_markov_property} and again below in the optional stopping theorems.
\end{properties}
\begin{usedin} The dyadic approximants $T_n\downarrow T$ are exactly how the discrete-time optional stopping theorem is bootstrapped up to continuous time in \cref{chap3.4:theorem-optional_stopping_theorem_for_martingales_UI_criteria,chap3.4:proposition-optional_stopping_nonnegative_supermartingales} below. \end{usedin}

We now record two closure properties of progressive processes under stopping, needed throughout the rest of this chapter (and, e.g., Definition \ref{chap4.1:definition-continuous_fv_process} in Chapter 4) to freely write down objects like $X_T$ or $X^T$ without re-checking measurability by hand every time.

\begin{proposition}[$X_T$ is $\mc F_T$-measurable, and stopping preserves progressivity] \label{chap3.2:proposition-progressive_at_stopping_time}
Let $(X_t)_{t\ge0}$ be progressive with values in $(E,\mc E)$ and let $T$ be a stopping time.
\begin{enumerate}[(i)]
    \item \LG{Theorem 3.7} The map $\omega\mapsto X_T(\omega):=X_{T(\omega)}(\omega)$, defined on $\{T<\infty\}$, is $\mc F_T$-measurable.
    \item The stopped process $X^T=(X_{t\wedge T})_{t\ge0}$ is again progressive. (This is not separately numbered by Le Gall; it is an immediate consequence of unwinding \cref{chap3.1:definition-measurable_and_progressive_processes} as below, and is used silently throughout, e.g.\ to make sense of $X^T$ in \cref{chap3.4:corollary-stopped_martingale_is_martingale}.)
\end{enumerate}
\end{proposition}
\begin{proofidea}
(i) Fix $t\ge0$ and work on $\{T\le t\}$, where $\omega\mapsto X_T(\omega)$ factors as
$$\omega\longmapsto(\omega,T(\omega)\wedge t)\longmapsto X_{T(\omega)\wedge t}(\omega).$$
The first arrow is $\mc F_t$-measurable because $T\wedge t$ is $\mc F_t$-measurable (\cref{chap3.2:proposition-stopping_time_of_right_cont_filtration}); the second is the restriction of $X$ to $\Omega\times[0,t]$, which is $\mc F_t\otimes\mc B([0,t])$-measurable by progressivity. The composite is therefore $\mc F_t$-measurable on $\{T\le t\}$ for every $t\ge0$, and \cref{chap3.2:properties-algebra_of_stopping_times} (last item) turns this into $\mc F_T$-measurability of $X_T$ on $\{T<\infty\}$.

(ii) Fix $t\ge0$; we check progressivity of $X^T$ directly from \cref{chap3.1:definition-measurable_and_progressive_processes}, i.e.\ that $(\omega,s)\mapsto X_{s\wedge T(\omega)}(\omega)$ is $\mc F_t\otimes\mc B([0,t])$-measurable on $\Omega\times[0,t]$. This map factors as
$$(\omega,s)\longmapsto(\omega,\,s\wedge T(\omega)\wedge t)\longmapsto X_{s\wedge T(\omega)\wedge t}(\omega),$$
and both arrows are measurable for the same reason as in (i): $T\wedge t$ is $\mc F_t$-measurable, so the first map is $\mc F_t\otimes\mc B([0,t])$-measurable, and the second is again just the restriction of $X$ to $\Omega\times[0,t]$. Hence $X^T$ satisfies the defining progressivity criterion at every $t$.
\end{proofidea}

\begin{proposition}[Hitting times] \label{chap3.2:proposition-hitting_times} \LG{Proposition 3.9} \\
Let $(X_t)_{t\ge0}$ be adapted with values in a metric space $(E,d)$.
\begin{enumerate}[(i)]
    \item If the sample paths of $X$ are right-continuous and $O\subseteq E$ is open, then $T_O=\inf\{t\ge0:X_t\in O\}$ is a stopping time of $(\mc F_{t+})$.
    \item If the sample paths of $X$ are continuous and $F\subseteq E$ is closed, then $T_F=\inf\{t\ge0:X_t\in F\}$ is a stopping time of $(\mc F_t)$.
\end{enumerate}
\end{proposition}
\begin{proofidea}
(i) By right-continuity, $\{T_O<t\}=\bigcup_{s\in[0,t)\cap\mathbb Q}\{X_s\in O\}$ (openness of $O$ lets any path visiting $O$ on $[0,t)$ be caught at a nearby rational time), which lies in $\mc F_t$; conclude with \cref{chap3.2:proposition-stopping_time_of_right_cont_filtration}. (ii) By continuity, the distance function $s\mapsto d(X_s,F)$ is continuous, so its infimum over $[0,t]$ already equals its infimum over the countable set $[0,t]\cap\mathbb Q$, giving $\{T_F\le t\}=\{\inf_{0\le s\le t}d(X_s,F)=0\}=\{\inf_{s\in[0,t]\cap\mathbb Q}d(X_s,F)=0\}\in\mc F_t$ directly --- no detour through $\mc F_{t+}$ is needed, which is why continuity (rather than mere right-continuity) buys a genuine $(\mc F_t)$-stopping time.
\end{proofidea}
\begin{quizbox} Which hypothesis buys which stopping time: right-continuous paths + \emph{open} target set gives a stopping time of $(\mc F_{t+})$ only (via $\{T<t\}\in\mc F_t$, \cref{chap3.2:proposition-stopping_time_of_right_cont_filtration}); continuous paths + \emph{closed} target set upgrades this to a genuine $(\mc F_t)$-stopping time (via $\{T\le t\}\in\mc F_t$ directly). If $(\mc F_t)$ is already right-continuous the two notions coincide, which is the main structural reason the ``usual conditions'' are assumed throughout. Separately: for progressive $X$, $X_T\ind_{\{T<\infty\}}$ is always $\mc F_T$-measurable (\cref{chap3.2:proposition-progressive_at_stopping_time}) --- this is the fact that lets one freely write $X_T$, $B_{T_a}$, etc.\ and treat them as $\mc F_T$-measurable random variables without further comment. \end{quizbox}





\subsection{Continuous-time martingales and supermartingales}
From here on all processes take values in $\R$. The definitions below are the direct continuous-time transcription of the discrete-time notions (treated separately in the appendix); what is genuinely new to this section is the sample-path regularity that right-continuity of $(\mc F_t)$ buys for free.

\begin{definition}[(Sub/super)martingale] \label{chap3.3:definition-sub_super_martingale} \LG{Definition 3.10} \\
An adapted process $(X_t)_{t\ge0}$ with $X_t\in L^1$ for every $t$ is a
\begin{itemize}
    \item \textit{martingale} if $E[X_t\mid\mc F_s]=X_s$ for every $0\le s<t$;
    \item \textit{supermartingale} if $E[X_t\mid\mc F_s]\le X_s$ for every $0\le s<t$;
    \item \textit{submartingale} if $E[X_t\mid\mc F_s]\ge X_s$ for every $0\le s<t$.
\end{itemize}
$X$ is a submartingale iff $-X$ is a supermartingale, so every statement below for supermartingales has an immediate submartingale counterpart. Taking expectations, $t\mapsto E[X_t]$ is constant (martingale), non-increasing (supermartingale), or non-decreasing (submartingale). The basic example is a \textit{Doob martingale} $X_t=E[Z\mid\mc F_t]$ for fixed $Z\in L^1$; not every martingale arises this way (e.g.\ $X_t=B_t$ is not of the form $E[Z\mid\mc F_t]$ for any single $Z$, since $B_t\not\to$ any $L^1$ limit as $t\to\infty$ --- see \cref{chap2.1:remark-time_inversion}).
\end{definition}

\begin{definition}[$(\mc F_t)$-Brownian motion] \label{chap3.3:definition-ft_brownian_motion} \LG{Definition 3.11} \\
A Brownian motion $B$ (\cref{chap2.2:definition-brownian_motion}) is an \textit{$(\mc F_t)$-Brownian motion} if it is adapted to $(\mc F_t)$ and has independent increments with respect to $(\mc F_t)$, i.e.\ $B_t-B_s\perp\mc F_s$ for every $s<t$ (not merely independent of $B_r$, $r\le s$). Every Brownian motion is an $(\mc F_t^B)$-Brownian motion for its own (possibly completed) canonical filtration, by \cref{chap2.1:proposition-characterisations_of_brownian_motion}.
\end{definition}

\begin{example}[Independent increments, and the Brownian case] \label{chap3.3:example-brownian_martingales} \LG{p.\ 49--51} \\
Suppose $(Z_t)_{t\ge0}$ is adapted with \textit{independent increments}: $Z_t-Z_s\perp\mc F_s$ for every $s<t$ (e.g.\ an $(\mc F_t)$-Brownian motion above). Writing $\hat Z_t:=Z_t-E[Z_t]$, each of the following is a martingale whenever it is well defined:
\begin{enumerate}[(i)]
    \item $\hat Z_t$, if $Z_t\in L^1$ for every $t$;
    \item $\hat Z_t^2-E[\hat Z_t^2]$, if $Z_t\in L^2$ for every $t$ (expand $\hat Z_t=\hat Z_s+(Z_t-Z_s)$ and use that the cross term has mean $0$ given $\mc F_s$, by independence of the increment);
    \item $X_t=e^{\theta Z_t}/E[e^{\theta Z_t}]$, for any fixed $\theta$ with $E[e^{\theta Z_t}]<\infty$ for every $t$ (immediate from independence of $Z_t-Z_s$ and $\mc F_s$).
\end{enumerate}
Specialising to $Z=B$ an $(\mc F_t)$-Brownian motion from $0$ recovers the three families
$$B_t, \qquad B_t^2-t, \qquad \exp\Big(\theta B_t-\frac{\theta^2}{2}t\Big),\qquad \theta\in\R,$$
all with continuous sample paths; the last family is the \textit{exponential martingale} of $B$, central to the Girsanov-type arguments of Chapter 5. The same recipe applied to a Wiener integral $Z_t=\int_0^tf(s)\,dB_s$ ($f\in L^2(\R_+,dt)$) gives martingales $Z_t$, $Z_t^2-\int_0^tf^2\,ds$, $\exp(\theta Z_t-\tfrac{\theta^2}2\int_0^tf^2\,ds)$ --- these too admit continuous modifications, by the general construction of Chapter 5.
\end{example}
\begin{trap}
Independence of increments alone is not enough to get a \emph{continuous} modification: a Poisson process $N$ with rate $\lambda$ also has independent increments, so $N_t-\lambda t$, $(N_t-\lambda t)^2-\lambda t$ and $\exp(\theta N_t-\lambda t(e^\theta-1))$ are martingales by exactly the same computation --- but none of them has a continuous modification, since $N$ itself jumps. Continuity of the exponential martingale above is a genuine extra feature of Brownian motion, not a free consequence of the independent-increments construction.
\end{trap}

\begin{proposition}[Doob's maximal and $L^p$ inequalities] \label{chap3.3:proposition-maximal_and_doob_Lp_inequality} \LG{Proposition 3.15} \\
\begin{enumerate}[(i)]
    \item If $X$ is a supermartingale with right-continuous paths, then for every $t>0$ and $\lambda>0$,
    $$\lambda\,P\Big(\sup_{0\le s\le t}|X_s|>\lambda\Big)\le E[|X_0|]+2E[|X_t|].$$
    \item If $X$ is a martingale with right-continuous paths, then for every $t>0$ and $p>1$,
    $$E\Big[\sup_{0\le s\le t}|X_s|^p\Big]\le\Big(\frac{p}{p-1}\Big)^pE[|X_t|^p].$$
\end{enumerate}
\end{proposition}
\begin{proofidea}
Fix a countable dense $D\subseteq\R_+$ with $0,t\in D$, and exhaust $D\cap[0,t]$ by an increasing sequence of finite grids $D_m=\{0=t_0^m<\dots<t_m^m=t\}$. On each $D_m$, $(X_{t_i^m})_i$ is a \emph{discrete-time} (super)martingale, so the discrete maximal/Doob inequalities apply verbatim and give the stated bounds with $\sup_{D_m}$ in place of $\sup_{[0,t]}$. Let $m\to\infty$: $\sup_{D_m}|X_s|\uparrow\sup_{D\cap[0,t]}|X_s|$, so monotone convergence (resp.\ Fatou for (i)) transfers the bounds to the sup over $D\cap[0,t]$. Finally, right-continuity of sample paths together with $t\in D$ gives $\sup_{D\cap[0,t]}|X_s|=\sup_{[0,t]}|X_s|$, since every point of $[0,t]$ is a decreasing limit of points of $D$.
\end{proofidea}

\begin{proposition}[Upcrossings, and existence of one-sided limits along a dense set] \label{chap3.3:proposition-upcrossing_and_onesided_limits} \LG{Lemma 3.16, combined with the upcrossing bound used in Theorem 3.17} \\
For $a<b$ and a real function $f$ on a set $I\subseteq\R_+$, let $M_{a,b}^f(I)\in\N\cup\{\infty\}$ be the largest $k$ such that $f$ crosses from $\le a$ to $\ge b$ at least $k$ times on $I$ (formally: there exist $s_1<t_1<\dots<s_k<t_k$ in $I$ with $f(s_i)\le a\le b\le f(t_i)$ for every $i$). If $X$ is a supermartingale and $D$ is a countable dense subset of $\R_+$, the discrete-time Doob upcrossing inequality applied along finite grids exhausting $D\cap[0,T]$ (exactly as in \cref{chap3.3:proposition-maximal_and_doob_Lp_inequality}) gives, for every rational $a<b$ and $T\in D$,
$$E\big[M_{a,b}^X(D\cap[0,T])\big]\le\frac{1}{b-a}E[(X_T-a)^-]<\infty.$$
Consequently, almost surely, $M_{a,b}^X(D\cap[0,T])<\infty$ simultaneously for every rational $a<b$ and every $T\in D$. A path with finitely many upcrossings of every rational level, on every bounded subinterval of $D$, cannot oscillate: it has a right-limit $X_{t+}:=\lim_{s\downarrow t,\,s\in D}X_s$ at every $t\ge0$ and a left-limit $X_{t-}:=\lim_{s\uparrow t,\,s\in D}X_s$ at every $t>0$, and $t\mapsto X_{t+}$ is càdlàg.
\end{proposition}
\begin{proofidea}
The finite-upcrossing bound is the content cited above (a purely analytic fact about real sequences, not reproved here). Given it, suppose $f:D\to\R$ fails to have a right-limit at some $t$: then $\limsup_{s\downarrow t}f(s)>\liminf_{s\downarrow t}f(s)$, so $f$ crosses every level $a<b$ strictly between these two values infinitely often on $D\cap(t,t+\epsilon)$ for every $\epsilon>0$, i.e.\ $M_{a,b}^f(D\cap[0,T])=\infty$ for $T>t$ close to $t$ --- contradicting finiteness. The left-limit argument is symmetric, and càdlàg-ness of $t\mapsto X_{t+}$ is the easy bookkeeping fact that right-limits along a dense set, taken as a function of the base point, are automatically right-continuous with left-limits.
\end{proofidea}

\begin{theorem}[Regularisation: existence of a càdlàg modification] \label{chap3.3:theorem-cadlag_modification} \LG{Theorems 3.17--3.18} \\
Let $X$ be a supermartingale. Then, almost surely, $X_{t+}$ and $X_{t-}$ (\cref{chap3.3:proposition-upcrossing_and_onesided_limits}) exist simultaneously at every $t$, $X_{t+}\in L^1$, and
$$X_t\le E[X_{t+}\mid\mc F_t],$$
with equality if $t\mapsto E[X_t]$ is right-continuous (in particular if $X$ is a martingale); the process $(X_{t+})_{t\ge0}$ is itself a supermartingale (a martingale, if $X$ is) with respect to $(\mc F_{t+})$.

If moreover $(\mc F_t)$ satisfies the usual conditions (complete and right-continuous) and $t\mapsto E[X_t]$ is right-continuous, then $X$ has a modification with càdlàg sample paths, which is again an $(\mc F_t)$-supermartingale.
\end{theorem}
\begin{proofidea}
\textit{Existence of $X_{t+}$ a.s.:} off a single null set $N$ built from \cref{chap3.3:proposition-upcrossing_and_onesided_limits} (ranging over rational $a<b$ and $T\in D$), every path has right- and left-limits along $D$.

\textit{The inequality and $L^1$ membership:} fix $t$ and a sequence $t_n\downarrow t$ in $D$; running time \emph{backwards} along $(t_n)$ turns $(X_{t_n})_n$ into a \textit{backward} discrete supermartingale (indexed by $\mc F_{t_n}\supseteq\mc F_{t_{n+1}}\supseteq\dots$), which by \cref{chap3.3:proposition-maximal_and_doob_Lp_inequality}-type uniform-$L^1$-bound considerations and the backward martingale convergence theorem converges not just a.s.\ but in $L^1$ to $X_{t+}$. $L^1$-convergence (not merely a.s.) is exactly what lets us pass to the limit inside the conditional expectation in $X_t\le E[X_{t_n}\mid\mc F_t]$ to get $X_t\le E[X_{t+}\mid\mc F_t]$, and equality holds once $E[X_t]=\lim_nE[X_{t_n}]=E[X_{t+}]$ forces equality in that conditional inequality.

\textit{Càdlàg modification:} set $Y_t=X_{t+}$ off the exceptional null set $N$ (and $Y_t=0$ on $N$); $Y$ is càdlàg by \cref{chap3.3:proposition-upcrossing_and_onesided_limits}. Right-continuity of $(\mc F_t)$ makes the $\mc F_{t+}$-measurable $X_{t+}$ already $\mc F_t$-measurable, and completeness absorbs the null set $N\in\mc F_\infty$ into $\mc F_0$, so $Y$ is adapted to $(\mc F_t)$. Finally $X_t=E[X_{t+}\mid\mc F_t]=X_{t+}=Y_t$ a.s.\ (using right-continuity of $t\mapsto E[X_t]$ for the first equality and $\mc F_t$-measurability of $X_{t+}$ for the second), so $Y$ is a modification of $X$ inheriting the supermartingale property.
\end{proofidea}
\begin{trap}
Both hypotheses are load-bearing. Without right-continuity of $(\mc F_t)$: take $\Omega=\{\pm1\}$ with $P(\pm1)=\tfrac12$, $\xi(\omega)=\omega$, and $X_t=0$ for $t\le1$, $X_t=\xi$ for $t>1$; $X$ is a martingale for its own canonical filtration (which has $\mc F_{1+}^X\ne\mc F_1^X$, so is \emph{not} right-continuous), yet no modification of $X$ can be right-continuous at $t=1$. Without right-continuity of $t\mapsto E[X_t]$: take $X_t=f(t)$ deterministic and non-increasing but not right-continuous; again no càdlàg modification exists.
\end{trap}

\begin{theorem}[A.s.\ convergence of $L^1$-bounded supermartingales] \label{chap3.3:theorem-as_convergence_L1_bounded} \LG{Theorem 3.19} \\
Let $X$ be a supermartingale with right-continuous sample paths such that $\sup_{t\ge0}E[|X_t|]<\infty$. Then $X_t\to X_\infty$ a.s.\ as $t\to\infty$, for some $X_\infty\in L^1$.
\end{theorem}
\begin{proofidea}
Exactly the upcrossing argument of \cref{chap3.3:proposition-upcrossing_and_onesided_limits}, but now letting $T\to\infty$ along $D$ as well: $L^1$-boundedness gives $\sup_TE[(X_T-a)^-]<\infty$, so $E[M_{a,b}^X(D)]<\infty$ for every rational $a<b$, ruling out infinite oscillation of the whole path (not just locally), so $X_t$ converges a.s.\ along $D$ to some limit in $[-\infty,\infty]$; Fatou's lemma rules out $\pm\infty$ and gives $X_\infty\in L^1$; right-continuity then removes the restriction to $t\in D$.
\end{proofidea}
\begin{remark}
This a.s.\ limit need not be an $L^1$ limit --- that refinement, and its equivalence with uniform integrability and with $X$ being \textit{closed}, is \cref{chap3.4:theorem-martingale_equivalence_of_L1_convergence} below, which is the martingale case ($X$ bounded in $L^1$ is then automatic from closedness).
\end{remark}
\begin{quizbox} Doob $L^2$: $E[\sup_{s\le t}X_s^2]\le4E[X_t^2]$. UI $\iff$ closed martingale. \end{quizbox}

\subsection{Optional stopping theorems}\label{sec:ost}
Having a candidate limit $X_\infty$ for a martingale (\cref{chap3.3:theorem-as_convergence_L1_bounded}) lets us finally make sense of $X_T$ for \emph{any} stopping time $T$, including $T=\infty$: once $X_t\to X_\infty$ a.s., set $X_T:=\ind_{\{T<\infty\}}X_T+\ind_{\{T=\infty\}}X_\infty$. This agrees with \cref{chap3.2:proposition-progressive_at_stopping_time} on $\{T<\infty\}$, and $X_T$ remains $\mc F_T$-measurable on all of $\Omega$ (the extra piece $\ind_{\{T=\infty\}}X_\infty$ is $\mc F_T$-measurable since $\{T=\infty\}\in\mc F_T$ and $X_\infty\in\mc F_\infty$, intersected appropriately). We call a martingale $X$ \textit{closed} if $X_t=E[Z\mid\mc F_t]$ for every $t$, for some fixed $Z\in L^1$ --- i.e.\ it is a Doob martingale in the sense of \cref{chap3.3:definition-sub_super_martingale}.

\begin{theorem}[Closed $\iff$ UI $\iff$ $L^1$-convergent] \label{chap3.4:theorem-martingale_equivalence_of_L1_convergence} \LG{Theorem 3.21} \\
Let $X$ be a martingale with right-continuous sample paths. The following are equivalent:
\begin{enumerate}[(i)]
    \item $X$ is closed;
    \item $(X_t)_{t\ge0}$ is uniformly integrable;
    \item $X_t$ converges a.s.\ \emph{and} in $L^1$ as $t\to\infty$.
\end{enumerate}
When they hold, writing $X_\infty$ for the common limit, $X_t=E[X_\infty\mid\mc F_t]$ for every $t\ge0$.
\end{theorem}
\begin{proofidea}
(i)$\Rightarrow$(ii): the family $\{E[Z\mid\mc G]:\mc G\text{ a sub-}\sigma\text{-field}\}$ is uniformly integrable for fixed $Z\in L^1$ --- a standard closure property of conditional expectations, not special to this setting.
(ii)$\Rightarrow$(iii): uniform integrability gives $L^1$-boundedness, so \cref{chap3.3:theorem-as_convergence_L1_bounded} gives a.s.\ convergence to some $X_\infty$; uniform integrability upgrades a.s.\ convergence to $L^1$ convergence.
(iii)$\Rightarrow$(i): pass to the limit $t\to\infty$ in $X_s=E[X_t\mid\mc F_s]$ (valid since conditional expectation is $L^1$-continuous) to get $X_s=E[X_\infty\mid\mc F_s]$, i.e.\ $X$ is closed by $Z=X_\infty$.
\end{proofidea}

\begin{theorem}[Optional stopping, UI case] \label{chap3.4:theorem-optional_stopping_theorem_for_martingales_UI_criteria} \LG{Theorem 3.22} \\
Let $X$ be a uniformly integrable martingale with right-continuous sample paths, and let $S\le T$ be stopping times. Then $X_S,X_T\in L^1$ and
$$X_S=E[X_T\mid\mc F_S].$$
In particular $X_S=E[X_\infty\mid\mc F_S]$ for every stopping time $S$, and $E[X_S]=E[X_\infty]=E[X_0]$.
\end{theorem}
\begin{proofidea}
Discretise with the dyadic approximants $S_n\downarrow S$, $T_n\downarrow T$ of \cref{chap3.2:properties-algebra_of_stopping_times} (note $S_n\le T_n$ since $S\le T$). At fixed $n$, $2^nS_n,2^nT_n$ are \emph{discrete} stopping times for the discrete filtration $(\mc F_{k2^{-n}})_k$, and $(X_{k2^{-n}})_k$ is a discrete UI martingale, so the discrete optional stopping theorem gives $X_{S_n}=E[X_{T_n}\mid\mc F_{S_n}]$ directly. Right-continuity of paths gives $X_{S_n}\to X_S$ and $X_{T_n}\to X_T$ a.s.; the discrete theorem applied again shows $X_{S_n}=E[X_\infty\mid\mc F_{S_n}]$ for every $n$, so $(X_{S_n})_n$ is itself uniformly integrable, upgrading the a.s.\ convergence to $L^1$. Passing to the $L^1$-limit in $E[\ind_AX_{S_n}]=E[\ind_AX_{T_n}]$ for $A\in\mc F_S\subseteq\mc F_{S_n}$ gives $E[\ind_AX_S]=E[\ind_AX_T]$ for every $A\in\mc F_S$, which is the claim.
\end{proofidea}

\begin{theorem}[Optional stopping, bounded stopping times] \label{chap3.4:theorem-optional_stopping_theorem_for_martingales_bounded_stopping_times} \LG{Theorem 3.23} \\
Let $X$ be a martingale with right-continuous sample paths (not assumed UI), and let $S\le T$ be \emph{bounded} stopping times, say $S,T\le a$. Then $X_S,X_T\in L^1$ and $X_S=E[X_T\mid\mc F_S]$.
\end{theorem}
\begin{proofidea}
Boundedness is exactly what lets us discard the UI hypothesis: the process $X^a=(X_{t\wedge a})_{t\ge0}$ is closed by $X_a\in L^1$ (take $Z=X_a$), hence UI by \cref{chap3.4:theorem-martingale_equivalence_of_L1_convergence}. Since $S,T\le a$, $X_S=X^a_S$ and $X_T=X^a_T$, so \cref{chap3.4:theorem-optional_stopping_theorem_for_martingales_UI_criteria} applied to $X^a$ gives the result.
\end{proofidea}

\begin{corollary}[The stopped process is again a martingale] \label{chap3.4:corollary-stopped_martingale_is_martingale} \LG{Corollary 3.24} \\
Let $X$ be a martingale with right-continuous sample paths and $T$ a stopping time.
\begin{enumerate}[(i)]
    \item $X^T=(X_{t\wedge T})_{t\ge0}$ is again a martingale.
    \item If moreover $X$ is uniformly integrable, then $X^T$ is a uniformly integrable martingale, and $X_{t\wedge T}=E[X_T\mid\mc F_t]$ for every $t\ge0$.
\end{enumerate}
\end{corollary}
\begin{proofidea}
(ii): $t\wedge T$ is a stopping time (\cref{chap3.2:properties-algebra_of_stopping_times}), so \cref{chap3.4:theorem-optional_stopping_theorem_for_martingales_UI_criteria} applied to $S=t\wedge T\le T$ gives $X_{t\wedge T}=E[X_T\mid\mc F_{t\wedge T}]=E[X_T\mid\mc F_t]$ for $A\in\mc F_t$, splitting the test set as $A\cap\{T\le t\}$ (where $X_{t\wedge T}=X_T$ trivially) and $A\cap\{T>t\}\in\mc F_t\cap\mc F_T=\mc F_{t\wedge T}$ (using $\mc F_{t\wedge T}=\mc F_t\cap\mc F_T$, \cref{chap3.2:properties-algebra_of_stopping_times}) to reduce to the defining property of $\mc F_{t\wedge T}$-conditional expectation. (i) follows by applying (ii) to the UI martingale $X^a=(X_{t\wedge a})_{t\ge0}$ for each fixed $a\ge0$, exactly as in \cref{chap3.4:theorem-optional_stopping_theorem_for_martingales_bounded_stopping_times}.
\end{proofidea}

\begin{proposition}[Supermartingale version] \label{chap3.4:proposition-optional_stopping_nonnegative_supermartingales} \LG{Theorem 3.25} \\
Let $Z$ be a nonnegative supermartingale with right-continuous sample paths (automatically bounded in $L^1$, hence a.s.\ convergent to some $Z_\infty$ by \cref{chap3.3:theorem-as_convergence_L1_bounded}). Let $U\le V$ be stopping times. Then $Z_U,Z_V\in L^1$ and
$$Z_U\ge E[Z_V\mid\mc F_U].$$
\end{proposition}
\begin{proofidea}
First for \emph{bounded} $U\le V\le p$: discretise exactly as in \cref{chap3.4:theorem-optional_stopping_theorem_for_martingales_UI_criteria} with bounded dyadic approximants $U_n\downarrow U$, $V_n\downarrow V$; the discrete optional stopping theorem for supermartingales gives $E[Z_{U_n}]\ge E[Z_{V_n}]$ and $Z_{U_n}\ge E[Z_{V_n}\mid\mc F_{U_n}]$ at each $n$, and a \emph{backward} supermartingale argument (as in \cref{chap3.3:theorem-cadlag_modification}) upgrades the a.s.\ convergence $Z_{U_n}\to Z_U$, $Z_{V_n}\to Z_V$ to $L^1$-convergence, so the inequality passes to the limit. For general (possibly unbounded or infinite) $U\le V$: apply the bounded case to $U\wedge p\le V\wedge p$ and let $p\to\infty$, using Fatou's lemma for integrability and the indicator device $U^A$ of \cref{chap3.2:properties-algebra_of_stopping_times}-style localisation (test against $A\in\mc F_U$ intersected with $\{U\le p\}$ and $\{U>p\}\subseteq\{V>p\}$ separately, then let $p\to\infty$) to recover the full inequality on $\mc F_U$.
\end{proofidea}

\begin{example}[Applications to Brownian hitting and exit times] \label{chap3.4:example-ost_applications} \LG{pp.\ 62--64} \\
Let $B$ be a real Brownian motion from $0$ and, for $a\in\R$, $T_a=\inf\{t\ge0:B_t=a\}$ (a.s.\ finite, \cref{chap2.3:proposition-sup_inf_and_hitting_times}).
\begin{enumerate}[(i)]
    \item \textit{Exit of an interval.} For $a,b>0$, let $T=T_{-a}\wedge T_b$ be the exit time of $(-a,b)$. Applying \cref{chap3.4:corollary-stopped_martingale_is_martingale} to the martingale $B$ --- the stopped process $B^T$ is bounded by $a\vee b$, hence UI --- gives $0=E[B_0]=E[B_T]=bP(T_b<T_{-a})-aP(T_{-a}<T_b)$, and together with $P(T_{-a}<T_b)+P(T_b<T_{-a})=1$ this yields
    $$P(T_{-a}<T_b)=\frac{b}{a+b}.$$
    Applying the same corollary to $M_t=B_t^2-t$ (also a martingale, \cref{chap3.3:example-brownian_martingales}) gives $E[B_{t\wedge T}^2]=E[t\wedge T]$; letting $t\to\infty$ (monotone convergence on the right, dominated convergence on the left since $B_{t\wedge T}^2\le a^2\vee b^2$) gives
    $$E[T]=ab.$$
    \item \textit{Laplace transform of $T_a$, $a>0$.} For $\lambda>0$, set $\theta=\sqrt{2\lambda}>0$ and apply \cref{chap3.4:corollary-stopped_martingale_is_martingale} to the exponential martingale $N^\theta_t=\exp(\theta B_t-\tfrac{\theta^2}2t)$ stopped at $T_a$: $N^\theta_{t\wedge T_a}$ is bounded above by $e^{\theta a}$ (since $\theta>0$ and $B_{s\wedge T_a}\le a$), hence UI, and $E[N^\theta_{T_a}]=E[N^\theta_0]=1$, i.e.\ $e^{\theta a}E[e^{-\lambda T_a}]=1$, giving
    $$E[e^{-\lambda T_a}]=e^{-a\sqrt{2\lambda}}.$$
\end{enumerate}
\end{example}
\begin{quizbox} Every application in the example: be fast at choosing the right martingale and justifying the limit (dominated/bounded convergence). \end{quizbox}
\begin{trap} $E[B_{T_a}]=a\ne0$: OST fails without UI/boundedness. \end{trap}
