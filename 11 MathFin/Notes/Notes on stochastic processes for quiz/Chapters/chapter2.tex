\subsection{Pre-Brownian motion}
\begin{definition}[Pre-Brownian motion] \label{chap2.1:definition-pre_brownian_motion} \LG{Definition 2.1} \\
Let $G$ be a Gaussian white noise on $\R_+$ with intensity equal to Lebesgue measure (\cref{chap1.4:definition-gaussian_white_noise}). The process $(B_t)_{t\ge0}$ defined by
$$B_t=G([0,t])$$
is called \textit{pre-Brownian motion}.
\end{definition}

\begin{proposition}[Characterisations] \label{chap2.1:proposition-characterisations_of_brownian_motion} \LG{Propositions 2.2--2.3} \\
Let $(X_t)_{t\ge0}$ be a real-valued random process. The following properties are equivalent:
\begin{enumerate}[(i)]
    \item $(X_t)_{t\ge0}$ is a pre-Brownian motion;
    \item $(X_t)_{t\ge0}$ is a centred Gaussian process with covariance $\Cov{X_s}{X_t}=s\wedge t$;
    \item $X_0=0$ a.s., and for every $0\le s<t$, the variable $X_t-X_s$ is independent of $\sigma(X_r,r\le s)$ and distributed according to $\Norm{0}{t-s}$;
    \item $X_0=0$ a.s., and for every choice of $0=t_0<t_1<\dots<t_p$, the increments $X_{t_i}-X_{t_{i-1}}$, $1\le i\le p$, are independent, with $X_{t_i}-X_{t_{i-1}}\sim\Norm{0}{t_i-t_{i-1}}$ for every $i$.
\end{enumerate}
\end{proposition}
\begin{proofidea}
(i)$\Rightarrow$(ii): by the defining isometry of $G$, the $B_t$ all lie in the common Gaussian space generated by $G$, and $E[B_sB_t]=E[G([0,s])G([0,t])]=\int_0^\infty\ind_{[0,s]}(r)\ind_{[0,t]}(r)\,dr=s\wedge t$.

(ii)$\Rightarrow$(iii): Let $H$ be the Gaussian space generated by $X$, let $H_s=\overline{\mathrm{span}}(X_r,r\le s)\subseteq H$ and $\tilde H_s=\overline{\mathrm{span}}(X_{s+u}-X_s,u\ge0)\subseteq H$. For $r\le s$ and $u\ge0$,
$$E[X_r(X_{s+u}-X_s)]=r\wedge(s+u)-r\wedge s=r-r=0,$$
so $H_s\perp\tilde H_s$, and \cref{chap1.3:theorem-independence_iff_orthogonality} gives $\sigma(H_s)\perp\sigma(\tilde H_s)$, i.e.\ $X_t-X_s$ is independent of $\sigma(X_r,r\le s)$ for $t>s$; its law follows directly from the covariance function.

(iii)$\Rightarrow$(iv): Taking $s=t_{p-1}$, $t=t_p$ in (iii) shows $X_{t_p}-X_{t_{p-1}}$ is independent of $(X_{t_1},\dots,X_{t_{p-1}})$; applying (iii) again with $s=t_{p-2}$, $t=t_{p-1}$ shows $X_{t_{p-1}}-X_{t_{p-2}}$ is independent of $(X_{t_1},\dots,X_{t_{p-2}})$, and so on, giving independence of all the increments.

(iv)$\Rightarrow$(i): (iv) shows that $X$ is a centred Gaussian process. For a step function $f=\sum_{i=1}^n\alpha_i\ind_{(t_{i-1},t_i]}$ on a subdivision $0=t_0<\dots<t_n$, set $G(f)=\sum_i\alpha_i(X_{t_i}-X_{t_{i-1}})$ (this depends only on $f$, not on the chosen subdivision representing it). Writing two step functions $f,g$ on a common subdivision and using (iv), a direct computation gives $E[G(f)G(g)]=\int_{\R_+}fg\,dr$. Since step functions are dense in $L^2(\R_+,\mathcal B(\R_+),dr)$, $f\mapsto G(f)$ extends to an isometry from $L^2(\R_+,dr)$ into the Gaussian space generated by $X$, i.e.\ to a Gaussian white noise with intensity Lebesgue measure, and $G([0,t])=X_t-X_0=X_t$.
\end{proofidea}

\begin{remark}[Finite-dimensional marginals, and recovering $G$ from $B$] \label{chap2.1:remark-fdd_and_white_noise_from_pre_bm} \LG{p.\ 21--22} \\
Property (iv) shows that the finite-dimensional marginals of pre-Brownian motion --- equivalently, the joint density of $(B_{t_1},\dots,B_{t_n})$ for $0=t_0<t_1<\dots<t_n$,
$$p(x_1,\dots,x_n)=\frac{1}{\sqrt{(2\pi)^nt_1(t_2-t_1)\cdots(t_n-t_{n-1})}}\exp\Big(-\sum_{i=1}^n\frac{(x_i-x_{i-1})^2}{2(t_i-t_{i-1})}\Big),\qquad x_0:=0,$$
--- completely determine the law of $B$; conversely, by property (iv) again, \emph{any} process with the same finite-dimensional marginals as pre-Brownian motion is itself a pre-Brownian motion.

The Gaussian white noise $G$ is in turn recovered from $B$ alone: writing $\int_0^\infty f(s)\,dB_s:=G(f)$ for $f\in L^2(\R_+,dt)$ (so $\int_u^v dB_s=G((u,v])=B_v-B_u$), the map $f\mapsto\int_0^\infty f(s)\,dB_s$ is called the \textit{Wiener integral} with respect to $B$, and $\int_0^\infty f(s)\,dB_s\sim\Norm{0}{\int_0^\infty f(s)^2\,ds}$. Since $G$ is not an ordinary ($\omega$-wise) measure, this is \emph{not} a pathwise integral; extending it to random integrands is exactly the programme carried out in Chapter 5 (\cref{sec:qv} onwards).
\end{remark}

\begin{proposition}[Invariance properties] \label{chap2.1:proposition-invariance_properties} \LG{Proposition 2.5} \\
Let $B$ be a pre-Brownian motion. Then:
\begin{enumerate}[(i)]
    \item $-B$ is also a pre-Brownian motion (\textit{symmetry});
    \item for every $\lambda>0$, the process $B^{(\lambda)}_t=\lambda^{-1}B_{\lambda^2t}$ is also a pre-Brownian motion (\textit{scaling});
    \item for every $s\ge0$, the process $B^{(s)}_t=B_{s+t}-B_s$ is a pre-Brownian motion independent of $\sigma(B_r,r\le s)$ (\textit{simple Markov property}).
\end{enumerate}
\end{proposition}
\begin{proofidea}
(i) and (ii) follow immediately by checking that $-B$, resp.\ $B^{(\lambda)}$, is a centred Gaussian process with covariance $s\wedge t$, and invoking \cref{chap2.1:proposition-characterisations_of_brownian_motion} (ii)$\Rightarrow$(i).
(iii) With the notation of the proof of \cref{chap2.1:proposition-characterisations_of_brownian_motion}, the $\sigma$-field generated by $B^{(s)}$ is $\sigma(\tilde H_s)$, which is independent of $\sigma(H_s)=\sigma(B_r,r\le s)$. To see that $B^{(s)}$ is itself a pre-Brownian motion it suffices to check property (iv) of \cref{chap2.1:proposition-characterisations_of_brownian_motion}, which is immediate since $B^{(s)}_{t_i}-B^{(s)}_{t_{i-1}}=B_{s+t_i}-B_{s+t_{i-1}}$.
\end{proofidea}

\begin{remark}[Time inversion] \label{chap2.1:remark-time_inversion} \LG{Exercise 2.25} \\
Let $W_0=0$ and $W_t=tB_{1/t}$ for $t>0$. Then $W$ is indistinguishable from a Brownian motion started at $0$. Indeed, for $s,t>0$,
$$\Cov{W_s}{W_t}=st\,\Cov{B_{1/s}}{B_{1/t}}=st\Big(\frac1s\wedge\frac1t\Big)=s\wedge t,$$
so $W$ is a pre-Brownian motion on $(0,\infty)$ by \cref{chap2.1:proposition-characterisations_of_brownian_motion}. On $(0,\infty)$, $t\mapsto W_t$ is continuous as a composition of continuous maps, so it only remains to show $W_t\to0$ a.s.\ as $t\downarrow0$. Since $W$ restricted to $\mathbb Q_+^*$ has the same law as $B$ restricted to $\mathbb Q_+^*$ (same fdds, by the computation above), and $B_q\to B_0=0$ a.s.\ along $q\in\mathbb Q_+^*\downarrow0$ by continuity of $B$, the same holds for $W$ along rationals, with probability one; combined with continuity on $(0,\infty)$ this gives $\lim_{t\downarrow0}W_t=0$ a.s.\ along \emph{all} $t\downarrow0$, not just rational ones. Consequently $W$ is indistinguishable from a Brownian motion started from $0$.

A direct consequence is the strong law of large numbers $B_t/t\to0$ a.s.\ as $t\to\infty$: applying this to $W$ and substituting $t=1/u$, $\lim_{u\to\infty}B_u/u=\lim_{t\downarrow0}W_t=0$ a.s.
\end{remark}

\subsection{Continuity of sample paths}
We start with a general definition of sample paths and what they mean. Let $E$ be a metric space, with metric $d$ equipped with its Borel $\sigma$-algebra.
\begin{definition}\label{chap2.2:definition-sample_paths_of_process}
\LG{Definition 2.6} \\
Let $(X_t)_{t \in T}$ be a random process indexed by $T$ with values in a metric space equipped with its Borel $\sigma$-algebra. The \textit{sample paths} of $X$ are the mappings $T \ni t \mapsto X_t(\omega)$ obtained when fixing $\omega \in \Omega$. The sample paths of $X$ thus form a collection of mappings from $T$ into $E$ indexed by $\omega \in \Omega$.    
\end{definition}
Let $B=(B_t)_{t \geq 0}$ be a pre-Brownian motion. At this present stage in Le Gall, we have no information about the sample paths of $B$: we cannot assert whether these are continuous or even measurable. By modifying the process $B$ we can ensure that sample paths are continuous.

\begin{definition}[Modification vs.\ indistinguishable]\label{chap2.2:definition-modification_and_indistinguishable}
\LG{Definition 2.7 and 2.8}\\
Let $X=(X_t)_{t \in T}$ and $\tilde{X}=(\tilde{X}_t)_{t \in T}$ be two random processes indexed by the same index set $T$ and with values in the same metric space $E$. We say that $\tilde{X}$ is a \textit{modification} of $X$ if
$$\forall t \in T: \quad P(\tilde{X}_t=X_t)=1$$
This implies that $\tilde{X}$ has the same finite-dimensional marginals as $X$\footnote{Thus, if $X$ is a pre-Brownian motion, $\tilde{X}$ is also a pre-Brownian motion. On the other hand, sample paths of $\tilde{X}$ may have very different properties from those of $X$. For instance, considering the case where $T = \R_+$ and $E=\R$, it is easy to construct examples where all the sample paths of $X$ are continuous whereas all sample paths of $\tilde{X}$ are discontinuous - one could for example let $\tilde{X}_t=X_t\cdot 1_{\mb{Q}}(t)$, where $X_t$ is some process that is almost surely not equal to 0, making it impossible for $\tilde{X}$ to be continuous}.
We say that the process $\tilde{X}$ is \textit{indistinguishable} from $X$ if there exists a negligible subset $N$ of $\Omega$ such that\footnote{One can also formulate this as $P(\forall t \in T: \tilde{X}_t = X_t)=1$, but the set may not be measurable in general, so this is slightly wrong.}
$$\forall \omega \in \Omega \setminus N, \forall t \in T: \quad \tilde{X}_t(\omega)=X_t(\omega).$$
\end{definition}
If $\tilde{X}$ is indistinguishable from $X$ then $\tilde{X}$ is a modification of $X$. The notion of indistinguishability is much stronger: Two indistinguishable processes have almost surely the same sample paths. Being a modification is not enough in general to imply indistinguishability: the null set where $\{\tilde{X}_t \neq X_t\}$ may depend on $t$, where as indistinguishability forces the null set where the processes do not almost surely agree to not depend on $t$. An uncountable infinite union of null sets may not in general be measurable. The following remark shows that continuity can be the missing piece
\begin{remark}\label{chap2.2:remark-continuity_makes_modifcation_equal_to_indistinguishability}
\LG{page 23}\\
Let $I$ be an interval of $\R$ and let $X=(X_t)_{t \in I}$ and $\tilde{X}=(\tilde{X}_t)_{t \in I}$ be random processes. Let $N$ and $\tilde{N}$ be negligible measurable sets of $\Omega$. If the sample paths of $X$ and $\tilde{X}$ are continuous on $\Omega \setminus N$ and $\Omega \setminus \tilde{N}$ respectively, then $\tilde{X}$ is a modification of $X$ if and only if $\tilde{X}$ is indistinguishable from $X$.    
\end{remark}
\begin{proof}
It is clear that indistinguishability implies a modification exists by the preceding observation.
If $\tilde{X}$ is a modification of $X$, we have almost surely that $X_t = \tilde{X}$ for every $t \in I \cap \mb{Q}$, because we can discard a countable union of negligible a countable union of negligible sets. Then, by a continuity argument, $X_t=\tilde{X}_t$ almost surely for every $t \in I$.
\end{proof}

\begin{theorem}[Kolmogorov's continuity criterion] \label{chap2.2:theorem-kolmogorovs_continuity_criterion}
\LG{Theorem 2.9} \\
Let $X=(X_t)_{t \in I}$ be a random process indexed by a bounded interval $I$ of $\R$, and taking values in a complete metric space $(E,d)$. Assume that there exists reals $q, \epsilon, C >0$ such that, for every $s,t \in I$,
\begin{equation}\label{chap2.2:equation-kolmogorovs_continuity_criterion}
E[d(X_s,X_t)^q] \leq C|t-s|^{1+\epsilon}.    
\end{equation}
Then, there exists a modification $\tilde{X}$ og $X$ whose sample paths are Hölder continuous with exponent $\alpha$ for every $\alpha \in (0, \frac{\epsilon}{q})$. This means that, for every $\omega \in \Omega$ and every $\alpha \in (0,\frac{\epsilon}{q})$, there exists a finite constant $C_\alpha(\omega)$ such that, for every $s,t \in I$,
$$d(\tilde{X}_s(\omega), \tilde{X}_t(\omega)) \leq C_\alpha(\omega)|t-s|^{\alpha}$$.
In particular, $\tilde{X}$ is a modification of $X$ with continuous sample paths, and by Remark \ref{chap2.2:remark-continuity_makes_modifcation_equal_to_indistinguishability} such a modification is unique up to indistinguishability.
\end{theorem}


Now we define the work-horse of the whole book: Brownian motion
\begin{definition}\label{chap2.2:definition-brownian_motion}
\LG{Definition 2.12} \\
A process $(B_t)_{t \geq0}$ is a (real) Brownian motion (started from 0) if:
\begin{enumerate}[(i)]
    \item $(B_t)_{t \geq 0}$ is a pre-Brownian motion.
    \item All sample paths of $B$ are continuous
\end{enumerate}
\end{definition}

\begin{corollary}\label{chap2.2:corollary-brownian_motion_exists}
\LG{Corollary 2.11} \\
Let $B = (B_t)_{t \geq 0}$ be a pre-Brownian, which exists by Proposition \ref{chap1.4:proposition-existence_of_gaussian_white_noise} using $f = 1_{[0,t]}$. Then Brownian motion exists in the following sense:
The process B has a modification whose sample paths are continuous, and even locally Hölder continuous with exponent $\frac{1}{2}-\delta$ for every $\delta \in (0,\frac{1}{2})$.
\end{corollary}
\begin{proof}
If $s<t$, then $B_t-B_s \sim \Norm{0,}{t-s}$, and thus $B_t-B_s$ has the same law as $\sqrt{t-s}U$, where $U \sim \Norm{0}{1}$. Consequently, for every $q >0$,
$$E[|B_t-B_s|^q]=(t-s)^{1/2}E[|U|^q]=C_q(t-s)^{q/2}$$
where $C_q=E[|U|^q]<\infty$. Taking $q > 2$ we can apply Kolmogorovs continuity criterion, Theorem \ref{chap2.2:theorem-kolmogorovs_continuity_criterion}, with $\epsilon = \frac{q}{2}-1$. It follows that $B$ has a modification (and hence is indistinguishable by Remark \ref{chap2.2:remark-continuity_makes_modifcation_equal_to_indistinguishability}, whose sample paths are locally Hölder continuous with exponent $\alpha$ for every $\alpha < \frac{1}{2}-\frac{1}{q}$. Taking $q$ arbitrarily large we can take $\alpha$ arbitrarily close to $\frac12$.
\end{proof}


\begin{definition}[Wiener measure, canonical construction] \label{chap2.2:definition-wiener_measure} \LG{pp.\ 27--28} \\
Let $C(\R_+,\R)$ be the space of continuous functions $\R_+\to\R$, equipped with the $\sigma$-field $\mathcal C$ generated by the coordinate maps $w\mapsto w(t)$, $t\ge0$ (equivalently, the Borel $\sigma$-field for the topology of uniform convergence on compacts). Given a Brownian motion $B$, the map $\omega\mapsto(t\mapsto B_t(\omega))$ from $\Omega$ to $C(\R_+,\R)$ is measurable, and the \textit{Wiener measure} $W$ is its image under $P$:
$$W(A)=P(B_\cdot\in A), \qquad A\in\mathcal C.$$
On a cylinder set $A=\{w:w(t_0)\in A_0,\dots,w(t_n)\in A_n\}$ with $0=t_0<\dots<t_n$ and $A_i\in\mathcal B(\R)$, $W(A)$ is given explicitly by the finite-dimensional density of \cref{chap2.1:remark-fdd_and_white_noise_from_pre_bm}. Since cylinder sets are stable under finite intersections and generate $\mathcal C$, a monotone class argument shows $W$ is determined by this formula alone --- in particular, $W$ does \emph{not} depend on the choice of Brownian motion $B$ used to define it: the law of Brownian motion is unique.

The \textit{canonical construction} takes $\Omega=C(\R_+,\R)$, $\mathcal F=\mathcal C$, $P=W$; the coordinate (``canonical'') process $X_t(w)=w(t)$ then has continuous sample paths by construction and the right fdds by the cylinder-set formula, hence is a Brownian motion on this space.
\end{definition}
\begin{quizbox} Kolmogorov's criterion (\cref{chap2.2:theorem-kolmogorovs_continuity_criterion}): $E[d(X_s,X_t)^q]\le C|t-s|^{1+\epsilon}$ on a bounded interval gives a modification that is Hölder continuous with exponent $\alpha$ for every $\alpha\in(0,\epsilon/q)$. For Brownian motion, Gaussian moments give $E[|B_t-B_s|^q]=C_q|t-s|^{q/2}$ for \emph{every} $q>0$, so every $q>2$ is admissible with $\epsilon=\frac q2-1$, yielding Hölder continuity for every $\alpha<\frac12-\frac1q$. Letting $q\to\infty$ exhausts every exponent $\alpha<\frac12$ --- but not $\alpha=\frac12$ itself, which genuinely fails (the law of the iterated logarithm shows $B$ is not $\frac12$-Hölder at any fixed point).
\end{quizbox}

\subsection{Properties of Brownian sample paths}
In this section, we investigate properties of sample paths of Brownian motion. We fix a Brownian motion $(B_t)_{t \geq 0}$. For every $t \geq 0$, we define the filtration like in Definition \ref{chap3.1:definition-filtration}
$$\mc{F}_t=\sigma(B_s, s \leq t)$$
We also set
$$\mc{F}_{0+}=\bigcap_{s > 0}\mc{F}_s$$

\begin{theorem}[Blumenthal's 0--1 law] \label{chap2.3:theorem-blumenthal_0_1_law}
\LG{Theorem 2.13} \\
The $\sigma$-algebra $\mc{F}_{0+}$ is trivial, in the sense that $P(A)\in \{0,1\}$ for every $A \in \mc{F}_{0+}$.
\end{theorem}

\begin{proposition}\label{chap2.3:proposition-sup_inf_and_hitting_times}
\LG{Proposition 2.14} \\
\begin{enumerate}
\item We have, a.s. for every $\epsilon>0$,
    $$\sup_{0 \leq s \leq \epsilon} B_s>0, \quad \inf_{0 \leq s \leq \epsilon} B_s<0$$
\item For every $a \in \R$, let $T_a=\inf\{t \geq 0|B_t=a\}$ (with the convention $\inf\emptyset=\infty$). Then,
$$\mathrm{a.s.}, \forall a \in \R: \quad T_a < \infty$$
Consequently, we have a.s.
$$\limsup_{t \to \infty}B_t=+\infty, \quad \liminf_{t \to \infty}B_t=-\infty$$
\end{enumerate}
\end{proposition}
\begin{remark}[Measurability of the running supremum] \label{chap2.3:remark-measurability_of_sup} \LG{p.\ 31} \\
It is not a priori clear that $\sup_{0\le s\le\epsilon}B_s$ is measurable, since this is an uncountable supremum of random variables. Continuity of sample paths lets us restrict the supremum to $s\in[0,\epsilon]\cap\mathbb Q$, a countable set, so it coincides with a supremum of countably many random variables and is therefore measurable. We use this kind of reduction implicitly throughout.
\end{remark}

\begin{corollary}\label{chap2.3:corollary-non_monotone_paths}
\LG{Corollary 2.15} \\
Almost surely, the function $t \mapsto B_t$ is not monotone on any non-trivial interval.
\end{corollary}



\begin{proposition}[Quadratic variation of BM] \label{chap2.3:proposition-quadratic_variation_of_brownian_motion} \LG{Prop 2.16} \\
Let $0 = t_0^n<t_1^n<...<t_{p_n}^n=t$ be a sequence of subdivisions of $[0,t]$ whose mesh tends to $0$ (i.e. $\sup_{1 \leq i \leq p_n}(t_i^n-t_{i-1}^n) \to 0$ as $n \to \infty$). Then
$$\lim_{n \to \infty} \sum_{i=1}^{p_n}(B_{t_i^n}-B_{t_{i-1}^n})^2=t$$
in $L^2$.
\end{proposition}

\begin{corollary}\label{chap2.3:corollary-infinite_variation}
\LG{Corollary 2.17} \\
Almost surely, the function $t \mapsto B_t$ has infinite variation on any non-trivial interval.
\end{corollary}

\subsection{Strong Markov property}
Our goal is to extend the simple Markov property (\cref{chap2.1:proposition-invariance_properties} (iii)) from a deterministic time $s$ to a \textit{random} time $T$. This requires restricting to the class of \textit{stopping times}, random times that can be recognised without looking into the future. The main payoff is the reflection principle below, which gives the exact joint law of $(S_t,B_t)$ where $S_t=\sup_{s\le t}B_s$; this is precisely the computation underlying the pricing of barrier options (knock-in/knock-out contracts whose payoff depends on whether the underlying --- modelled, after a change of measure and a log/drift transform, by a Brownian motion --- has crossed a fixed barrier by time $t$), and it reappears via hitting-time Laplace transforms in \cref{sec:ost}.

\begin{definition}[Stopping time; the $\sigma$-field $\mathcal F_T$] \label{chap2.4:definition-stopping_time} \LG{Definitions 2.18, 2.19} \\
(Preview of the systematic treatment of \cref{chap3.1:definition-filtration} onwards.) Recall $\mathcal F_t=\sigma(B_s,s\le t)$ and set $\mathcal F_\infty=\sigma(B_s,s\ge0)$. A random variable $T$ with values in $[0,\infty]$ is a \textit{stopping time} if
$$\forall t\ge0:\quad \{T\le t\}\in\mathcal F_t.$$
The $\sigma$-\textit{field of the past before} $T$ is
$$\mathcal F_T=\{A\in\mathcal F_\infty : \forall t\ge0,\; A\cap\{T\le t\}\in\mathcal F_t\}.$$
$\mathcal F_T$ is indeed a $\sigma$-field, $T$ is $\mathcal F_T$-measurable, and the random variable $\ind_{\{T<\infty\}}B_T$ is $\mathcal F_T$-measurable (approximate by the dyadic step variables $\ind_{\{T<\infty\}}\sum_i\ind_{\{i2^{-n}\le T<(i+1)2^{-n}\}}B_{i2^{-n}}$ and let $n\to\infty$).

\textit{Examples.} The constant $T=t$ and, for $a\in\R$, the hitting time $T_a=\inf\{t\ge0:B_t=a\}$ (\cref{chap2.3:proposition-sup_inf_and_hitting_times}) are stopping times, since $\{T_a\le t\}=\{\inf_{0\le s\le t}|B_s-a|=0\}\in\mathcal F_t$ by continuity of $B$. By contrast $L=\sup\{s\le1:B_s=0\}$ is \emph{not} a stopping time --- this follows from the strong Markov property below together with \cref{chap2.3:proposition-sup_inf_and_hitting_times}, since knowing $\{L\le t\}$ for $t<1$ would require knowing whether $B$ returns to $0$ on $(t,1]$, which is genuinely not $\mathcal F_t$-measurable information.
\end{definition}

\begin{theorem}[Strong Markov property] \label{chap2.4:theorem-strong_markov_property} \LG{Theorem 2.20} \\
Let $T$ be a stopping time with $P(T<\infty)>0$, and set, for every $t\ge0$,
$$B_t^{(T)}=\ind_{\{T<\infty\}}(B_{T+t}-B_T).$$
Then, under $P(\cdot\mid T<\infty)$, the process $(B_t^{(T)})_{t\ge0}$ is a Brownian motion independent of $\mathcal F_T$.
\end{theorem}
\begin{proofidea} \LG{pp.\ 34--35} \\
Assume first $T<\infty$ a.s. For $n\ge1$ let $[T]_n$ be the smallest number of the form $k2^{-n}$ that is $\ge T$, so $[T]_n\downarrow T$. Fix $A\in\mathcal F_T$, times $0\le t_1<\dots<t_p$, and bounded continuous $F:\R^p\to\R$. Since $A\cap\{(k-1)2^{-n}<T\le k2^{-n}\}$ is $\mathcal F_{k2^{-n}}$-measurable (it equals $(A\cap\{T\le k2^{-n}\})\cap\{T\le(k-1)2^{-n}\}^c$), the simple Markov property \ref{chap2.1:proposition-invariance_properties}(iii) applied at the deterministic time $k2^{-n}$ gives
\begin{align*}
&E\big[\ind_{A\cap\{(k-1)2^{-n}<T\le k2^{-n}\}}F(B_{k2^{-n}+t_1}-B_{k2^{-n}},\dots,B_{k2^{-n}+t_p}-B_{k2^{-n}})\big] \\
&\qquad=P\big(A\cap\{(k-1)2^{-n}<T\le k2^{-n}\}\big)\,E[F(B_{t_1},\dots,B_{t_p})].
\end{align*}
Summing over $k\ge0$ gives the identity with $[T]_n$ in place of $T$ on the left. As $n\to\infty$, $[T]_n\downarrow T$ and continuity of $B$ gives $B^{([T]_n)}_{t_i}\to B^{(T)}_{t_i}$ a.s.\ for each $i$, so dominated convergence (recall $F$ is bounded) transfers the identity to $T$:
$$E\big[\ind_AF(B^{(T)}_{t_1},\dots,B^{(T)}_{t_p})\big]=P(A)\,E[F(B_{t_1},\dots,B_{t_p})].$$
Taking $A=\Omega$ shows $B^{(T)}$ has the finite-dimensional marginals of a Brownian motion; its sample paths are continuous since $B$'s are; and the identity for general $A\in\mathcal F_T$ shows every finite-dimensional vector of $B^{(T)}$, hence $B^{(T)}$ itself, is independent of $\mathcal F_T$. The case $P(T=\infty)>0$ is identical after restricting everywhere to $\{T<\infty\}$.
\end{proofidea}

\begin{theorem}[Reflection principle] \label{chap2.4:theorem-reflection_principle} \LG{Theorem 2.21} \\
For $t>0$, let $S_t=\sup_{0\le s\le t}B_s$. Then, for every $a\ge0$ and $b\in(-\infty,a]$,
$$\PP(S_t\ge a, B_t\le b)=\PP(B_t\ge 2a-b).$$
In particular, $S_t\eqd|B_t|$.
\end{theorem}
\begin{proofidea} \LG{pp.\ 35--36} \\
Apply the strong Markov property at $T_a=\inf\{t\ge0:B_t=a\}$, which is a.s.\ finite by \cref{chap2.3:proposition-sup_inf_and_hitting_times}. On $\{T_a\le t\}=\{S_t\ge a\}$ we have $B_t-a=B^{(T_a)}_{t-T_a}$, and $(T_a,B^{(T_a)})$ has the same law as $(T_a,-B^{(T_a)})$, since $B^{(T_a)}$ is a Brownian motion independent of $T_a$ and $-B^{(T_a)}$ is again a Brownian motion with the same law. Rewriting $\{S_t\ge a,B_t\le b\}$ in terms of $(T_a,B^{(T_a)})$ and swapping the sign of $B^{(T_a)}$ therefore gives
\begin{align*}
\PP(S_t\ge a,B_t\le b) &=\PP\big(T_a\le t,\, B^{(T_a)}_{t-T_a}\le b-a\big)=\PP\big(T_a\le t,\,-B^{(T_a)}_{t-T_a}\le b-a\big) \\
&=\PP\big(T_a\le t,\,B_t\ge 2a-b\big)=\PP(B_t\ge2a-b),
\end{align*}
where the last equality holds because $b\le a$ forces $\{B_t\ge 2a-b\}\subseteq\{S_t\ge a\}=\{T_a\le t\}$ a.s. For the last claim, take $b=a$ and split
$$\PP(S_t\ge a)=\PP(S_t\ge a,B_t\ge a)+\PP(S_t\ge a,B_t\le a)=\PP(B_t\ge a)+\PP(B_t\ge a)=\PP(|B_t|\ge a),$$
using $\{B_t\ge a\}\subseteq\{S_t\ge a\}$ for the first term.
\end{proofidea}

\begin{corollary}[Law of $T_a$] \label{chap2.4:corollary-law_of_hitting_time} \LG{Corollary 2.22} \\
The pair $(S_t,B_t)$ has density
$$g(a,b)=\sqrt{\frac{2}{\pi t^3}}\,(2a-b)\exp\Big(-\frac{(2a-b)^2}{2t}\Big)\ind_{\{a>0,\,b<a\}}.$$
For every $a>0$, $T_a=\inf\{t\ge0:B_t=a\}$ has the same distribution as $a^2/B_1^2$, with density
$$f(t)=\frac{a}{\sqrt{2\pi t^3}}\exp\Big(-\frac{a^2}{2t}\Big)\ind_{\{t>0\}},$$
and in particular $E[T_a]=\infty$.
\end{corollary}
\begin{proofidea}
For every $t\ge0$, $\{T_a\le t\}=\{S_t\ge a\}$, and Theorem \ref{chap2.4:theorem-reflection_principle} gives $\PP(S_t\ge a)=\PP(|B_t|\ge a)=\PP(tB_1^2\ge a^2)$, i.e.\
$$\PP(T_a\le t)=\PP\Big(\frac{a^2}{B_1^2}\le t\Big).$$
Differentiating the right-hand side (a standard computation using the density of $B_1\sim\Norm01$) gives $f$. The density $g$ follows by differentiating Theorem \ref{chap2.4:theorem-reflection_principle} in $a,b$. Finiteness of $E[T_a]$ fails since $f(t)\sim a(2\pi)^{-1/2}t^{-3/2}$ as $t\to\infty$, which is not integrable.
\end{proofidea}

\begin{quizbox} Reflection principle: apply the strong Markov property at $T_a$; since $B^{(T_a)}$ is a Brownian motion independent of $T_a$, reflecting it ($B^{(T_a)}\mapsto-B^{(T_a)}$) leaves the joint law of $(T_a,B^{(T_a)})$ unchanged, which turns $\{S_t\ge a,B_t\le b\}$ into $\{B_t\ge2a-b\}$; the special case $b=a$ gives $S_t\eqd|B_t|$. Law of $T_a$: invert $\{T_a\le t\}=\{S_t\ge a\}=\{|B_t|\ge a\}$ to get $T_a\eqd a^2/B_1^2$, with density $\frac{a}{\sqrt{2\pi t^3}}e^{-a^2/2t}$ and $E[T_a]=\infty$ (heavy $t^{-3/2}$ tail). \end{quizbox}
